import { createRequire } from "node:module"; const { chromium } = createRequire("/home/user/ofis---y-netimi/package.json")("playwright");
const b = await chromium.launch();
const p = await b.newPage({ viewport: { width: 1080, height: 1350 }, deviceScaleFactor: 1 });
await p.goto("file://" + process.cwd() + "/tanitim.html");
await p.evaluate(() => document.fonts.ready);
await p.waitForTimeout(300);
const tasma = await p.evaluate(() => ({ w: document.documentElement.scrollWidth, h: document.documentElement.scrollHeight, listAlt: document.querySelector(".list").getBoundingClientRect().bottom, footUst: document.querySelector(".foot").getBoundingClientRect().top, fonts: [...document.fonts].filter(f => f.status === "loaded").map(f => f.family) }));
console.log(JSON.stringify(tasma));
await p.screenshot({ path: "DestekOfis-Tanitim-Mobil.png" });
await b.close();
