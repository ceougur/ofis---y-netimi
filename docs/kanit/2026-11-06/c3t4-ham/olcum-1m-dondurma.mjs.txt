#!/usr/bin/env node
// Ç3 tur 4 — ders 47 / 52 ölçümü: 1.000.000 verisinde (programın kendi ürettiği; kapi-olcum-yama.mjs --yalniz-kur; salt okunur kaynak, TAZE kopya) kilit 31.12.2025 + dondurmanın
// bitişi (yazım + tablo doğrulaması + tamamlama): süre ve "olay döngüsü en çok" (5 ms'lik yoklayıcı; pencereye yalnız pencere açıldıktan sonra başlayan boşluk yazılır), 50 ms üstü
// boşluk sayısı; /proc/loadavg başta, her adımda ve sonda. Kurulum dar kontrol p6 ile aynı (çekirdek kip: gerçek routes/ledger.mjs uçları + gerçek yönlendirici + gerçek HTTP;
// createApp YOK — 1 M açılışı Ç3 dışı AH bulgusu). Yük > 2 (4 çekirdek; ders 52) ise sonuç GEÇERSİZ yazılır.
// Kullanım: flock -x <olcum.lock> node olcum-1m-dondurma.mjs <kod kökü> <veri klasörü (n-1000000)> <çıktı.json>
import { copyFileSync, mkdtempSync, readFileSync, rmSync, writeFileSync } from "node:fs";
import { createServer } from "node:http";
import os from "node:os";
import path from "node:path";
import { pathToFileURL } from "node:url";

const [rootArg, veriArg, outArg] = process.argv.slice(2);
const root = path.resolve(rootArg);
const veri = path.resolve(veriArg);
const output = path.resolve(outArg);
const LOCK = "2025-12-31";
const NOW = "2026-10-08T12:00:00+03:00";
const LOAD_LIMIT = 2;
const imp = file => import(pathToFileURL(path.join(root, file)).href);
const loadavg = () => readFileSync("/proc/loadavg", "utf8").trim();
const work = mkdtempSync(path.join(path.dirname(output), "olcum-1m-"));
const sleep = ms => new Promise(resolve => setTimeout(resolve, ms));
const result = { kod: root, cpu: os.cpus().length, loadavgBasta: loadavg(), adimlar: [] };
let last = performance.now();
let windowMax = 0;
let gaps50 = 0;
const probe = setInterval(() => {
  const at = performance.now();
  const gap = at - last - 5;
  windowMax = Math.max(windowMax, gap);
  if (gap > 50) gaps50 += 1;
  last = at;
}, 5);
let server = null;
let db = null;
let snapshots = null;
try {
  const { createClock } = await imp("server/lib/clock.mjs");
  const { createStore, openDatabase } = await imp("server/lib/db.mjs");
  const { createPeriod } = await imp("server/lib/period.mjs");
  const { createRouter } = await imp("server/lib/router.mjs");
  const { HttpError, send } = await imp("server/lib/http.mjs");
  const { registerLedgerRoutes } = await imp("server/routes/ledger.mjs");
  const file = path.join(work, "destekofis.sqlite");
  copyFileSync(path.join(veri, "data", "destekofis.sqlite"), file);
  db = openDatabase(file);
  const store = createStore(db);
  const now = createClock({ time: NOW, fixed: true });
  const period = createPeriod({ store, now });
  if (period.lockedUntil()) throw new Error("veride kilit var");
  const router = createRouter();
  const admin = { id: "olcum", display_name: "Ölçüm", role: "admin", permissions: ["system.manage", "overview.view"] };
  let lockState = { lock: "", digest: "" };
  const gate = { adoptLock: ({ lock = "", digest = null } = {}) => (lockState = { lock, digest: lock ? digest : "" }), lockPending: lock => lockState.lock === lock && lockState.digest === null };
  const ledger = registerLedgerRoutes(router, { store, now, auth: { requirePermission: () => admin }, audit: () => {}, events: null, period, integrity: () => gate, money: null, config: { dbPath: file }, freezeBatchLines: 500 });
  snapshots = ledger.snapshots;
  server = createServer(async (req, res) => {
    const url = new URL(req.url, "http://127.0.0.1");
    try {
      const matched = router.match(req.method, url.pathname);
      if (!matched || matched.methodNotAllowed) throw new HttpError(404, "yok");
      await matched.handler({ req, res, url, params: matched.params });
    } catch (error) {
      if (res.headersSent) return res.end();
      if (error instanceof HttpError) return send(res, error.status, { ok: false, error: error.message, ...error.extra });
      send(res, 500, { ok: false, error: String(error?.message || error) });
    }
  });
  await new Promise(resolve => server.listen(0, "127.0.0.1", resolve));
  const base = `http://127.0.0.1:${server.address().port}`;
  const idle = () => !snapshots.status().running && !ledger.freezeView().active;
  const step = async (label, fn) => {
    await sleep(200);
    windowMax = 0;
    gaps50 = 0;
    const started = performance.now();
    const value = await fn();
    const row = { adim: label, ms: Math.round(performance.now() - started), olayDongusuEnCokMs: Math.round(windowMax), bosluk50msUstu: gaps50, ...value, loadavg: loadavg() };
    result.adimlar.push(row);
    console.log(JSON.stringify(row));
    return row;
  };
  const tally = () => {
    const cover = snapshots.coverage();
    const lastRun = store.get("SELECT no, state_lines FROM ledger_snapshot_runs WHERE state = 'done' ORDER BY no DESC LIMIT 1");
    return { kapsam: cover, gorunen: Number(store.get("SELECT COUNT(*) AS n FROM ledger_snapshots WHERE date <= ?", cover).n), kayitli: Number(lastRun?.state_lines ?? 0), hata: snapshots.status().error || "" };
  };
  result.loadavgOlcumOncesi = loadavg();
  await step("kilit 31.12.2025 (PUT yanıtı)", async () => {
    const started = performance.now();
    const response = await fetch(`${base}/api/admin/period-lock`, { method: "PUT", headers: { "content-type": "application/json" }, body: JSON.stringify({ lockedUntil: LOCK, force: true }) });
    const data = await response.json();
    return { status: response.status, yanitMs: Math.round(performance.now() - started), active: data?.data?.freeze?.active ?? null };
  });
  await step("dondurmanın bitişi (yazım + doğrulama + tamamlama)", async () => {
    for (let i = 0; i < 4 * 60 * 30 && !idle(); i += 1) await sleep(250);
    return tally();
  });
  const end = tally();
  result.SONUC = end.kapsam === LOCK && end.gorunen === end.kayitli && !end.hata ? "TUTARLI" : `HATA: ${JSON.stringify(end)}`;
} catch (error) {
  result.hata = String(error?.stack || error);
  console.log(result.hata);
} finally {
  clearInterval(probe);
  try {
    await new Promise(resolve => server?.close(resolve));
  } catch {
    // kapalı
  }
  try {
    await snapshots?.close?.();
  } catch {
    // kapalı
  }
  try {
    db?.close();
  } catch {
    // kapalı
  }
  rmSync(work, { recursive: true, force: true, maxRetries: 10, retryDelay: 200 });
}
result.loadavgSonda = loadavg();
// Geçerlilik ölçümden ÖNCEKİ yükle (başkalarının yükü): ölçüm sürerken 1 dk ortalamasına dondurmanın kendisi de (ana iş parçacığı + dondurma iş parçacığı ≈ 2) girer; o değerler
// yalnız yazılır (en çok değer ayrıca).
const start = Number(result.loadavgBasta.split(" ")[0]);
const peak = Math.max(start, ...result.adimlar.map(row => Number(String(row.loadavg).split(" ")[0])), Number(result.loadavgSonda.split(" ")[0]));
result.yukEnCok = peak;
result.GECERLILIK = start > LOAD_LIMIT ? `GEÇERSİZ: ölçümden önceki yük (1 dk ortalaması) ${start} > ${LOAD_LIMIT} (4 çekirdek; ders 52); ölçüm sırasında en çok ${peak}` : `GEÇERLİ: ölçümden önceki yük ${start} ≤ ${LOAD_LIMIT}; ölçüm sırasında en çok ${peak} (ölçümün kendi ~2 iş parçacığı dahil)`;
console.log(result.GECERLILIK);
writeFileSync(output, JSON.stringify(result, null, 1));
