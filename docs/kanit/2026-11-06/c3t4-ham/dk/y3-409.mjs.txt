// Temizlik "purge" kancasıyla TUTULUR (büyük şirket benzetimi; süreye bağlı değil). Her istek: durum kodu, kod, süre (ms), sonra kilit, koşu sayısı, kilit izi sayısı.
//   C1 kaldır → (temizlik sürerken) aynı güne yeniden kilit
//   C2 geri gün (03.31) → temizlik sürerken: ileri (31.12), eski güne (30.06), daha geri (31.01; temizlik istemez), sonra ileri (28.02)
//   C3 temizlik sürerken kapsamla aynı güne kilit (koşu açmaz)

// Kullanım: node y3-409.mjs <kod kökü> [senaryo …]
import { ENV, LOCK, MID, NOW, ROOT, imp, runsOf, seed, sleep, whole } from "./ortak.mjs";

const pick = process.argv.slice(3);
const { loginAdmin, removeTestDir, startTestServer } = await imp("test/helpers.mjs");
const { lockAndWait, waitForFreeze } = await imp("test/fixtures/donma-bekle.mjs");
const results = {};

function hooks() {
  const st = { purgeHold: false, batchHold: 0, batches: 0, purgeHeld: false, batchHeld: false };
  let releasePurge = () => {};
  let releaseBatch = () => {};
  let purgeGate = new Promise(resolve => (releasePurge = resolve));
  let batchGate = new Promise(resolve => (releaseBatch = resolve));
  const hook = {
    fn: async phase => {
      if (phase === "purge" && st.purgeHold) {
        st.purgeHeld = true;
        await purgeGate;
      }
      if (phase === "batch") {
        st.batches += 1;
        if (st.batchHold && st.batches === st.batchHold) {
          st.batchHeld = true;
          await batchGate;
        }
      }
    },
  };
  return {
    hook, st,
    openPurge: () => {
      st.purgeHold = false;
      releasePurge();
    },
    openBatch: () => {
      st.batchHold = 0;
      releaseBatch();
    },
  };
}
const lockSetting = store => store.get("SELECT value FROM settings WHERE key = 'ledger.lockedUntil'")?.value || "";
const lockTraces = store => store.get("SELECT COUNT(*) AS n FROM audit_events WHERE type IN ('ledger.period.locked', 'ledger.period.unlocked')").n;
async function timedPut(admin, store, day) {
  const t0 = performance.now();
  const response = await admin.put("/api/admin/period-lock", { lockedUntil: day });
  const ms = Math.round(performance.now() - t0);
  const data = response.data || {};
  return { gun: day || "kaldır", durum: response.status, kod: data.code || null, ileti: data.error || null, ms, sonra_kilit: lockSetting(store), kosu_sayisi: runsOf(store).length, kilit_izi: lockTraces(store), temizlik: (data.freeze || data.data?.freeze)?.cleanup || null };
}
async function until(cond, label, tries = 2000) {
  for (let i = 0; i < tries && !cond(); i += 1) await sleep(10);
  if (!cond()) throw new Error(`ön koşul oluşmadı: ${label}`);
}
async function scenario(name, body) {
  if (pick.length && !pick.includes(name)) return;
  const h = hooks();
  const server = await startTestServer({ now: { time: NOW, fixed: true }, env: ENV, __testFreezeHook: h.hook, freezeBatchLines: 5 });
  const out = { adimlar: [] };
  try {
    const admin = await loginAdmin(server);
    await seed(admin);
    await body({ server, admin, h, out, put: day => timedPut(admin, server.app.store, day) });
    h.openPurge();
    h.openBatch();
    await waitForFreeze(admin);
    out.kosular = runsOf(server.app.store).map(r => [r.no, r.kind, r.state, r.fromDate, r.through, r.rule, r.reason, r.lines]);
    out.butunluk = whole(server, lockSetting(server.app.store));
  } catch (error) {
    out.hata = String(error?.stack || error).slice(0, 1500);
  } finally {
    h.openPurge();
    h.openBatch();
    await server.app.close().catch(() => {});
    removeTestDir(server.root);
  }
  results[name] = out;
  console.error(`${name}: ${out.hata ? `HATA ${out.hata.split("\n")[0]}` : `${out.adimlar.map(a => `${a.gun}→${a.durum}${a.kod ? `/${a.kod}` : ""} ${a.ms}ms kilit=${a.sonra_kilit || "—"}`).join(" | ")} || tutarlı=${out.butunluk.tutarli} (${out.butunluk.gorunen}/${out.butunluk.kayitli}/${out.butunluk.canli}, kapsam ${out.butunluk.kapsam})`}`);
}

await scenario("C1", async ({ server, admin, h, out, put }) => {

  h.openPurge();
  await waitForFreeze(admin);
  out.adimlar.push(await put(LOCK));
});
