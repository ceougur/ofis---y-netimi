// Kullanım: node mutasyon.mjs <kopya kökü> <istemci|yenile|okuma>
import { readFileSync, writeFileSync } from "node:fs";
import path from "node:path";

const [root, kind] = process.argv.slice(2);
const swap = (rel, from, to) => {
  const file = path.join(root, rel);
  const text = readFileSync(file, "utf8");
  const count = text.split(from).length - 1;
  if (count !== 1) throw new Error(`${rel}: "${from.slice(0, 60)}" ${count} kez`);
  writeFileSync(file, text.replace(from, to));
  console.log(`değişti: ${rel}`);
};
const lineEdit = (rel, readLine, readText, prevLine, prevFrom, insert, prevTo) => {
  const file = path.join(root, rel);
  const lines = readFileSync(file, "utf8").split("\n");
  if (lines[readLine - 1].trim() !== readText) throw new Error(`${rel}:${readLine} beklenen değil: ${lines[readLine - 1]}`);
  if (!lines[prevLine - 1].includes(prevFrom)) throw new Error(`${rel}:${prevLine} beklenen değil: ${lines[prevLine - 1]}`);
  lines[prevLine - 1] = lines[prevLine - 1].replace(prevFrom, prevTo);
  lines.splice(readLine, 0, `    ${insert}`);
  writeFileSync(file, lines.join("\n"));
  console.log(`değişti: ${rel}`);
};
