async function measure({ version, dir, day }) {

  const { scoped } = await imp("test/guvenilirlik/m30-uretici.mjs");

  const edm = await startMockEdm();
  const started = performance.now();
  let server;
  try {
    server = await boot(version, dir, day);
  } catch (error) {
    await edm.close();
    throw error;
  }
  const result = { version: version === CURRENT ? "aday" : version, commit: server.commit, day, bootMs: Math.round(performance.now() - started), firstAccessMs: {}, companies: {}, files: {} };
  try {
    const api = await server.login();
    result.install = await installFacts(api);
    for (const company of result.install.companies) {
      const t = performance.now();
      const s = scoped(api, company.id);
      await s.get("/api/workspace/ledger/lock");
      result.firstAccessMs[company.code] = Math.round(performance.now() - t);
      result.companies[company.code] = await companyFacts(s);
    }
  } finally {
    await server.close();
    await edm.close();
  }
  const reg = readRegistry(path.join(dir, "data")) || [{ id: "sirket-001", dir: "" }];

    const d1 = stableDiff(taban.companies[code], old.companies[code]);
    check(`${code} göçten sonra ${OLD} açınca gösterdiği = göçten önce gösterdiği (${d1.map(x => x.section).join(",") || "fark yok"})`, !d1.length, { d1: JSON.stringify(d1).slice(0, 1500) });
    check(`${code} ${OLD} açılışı şemayı geri almaz (${SCHEMA} kalır; ${old.files[code].version}), mobil tablolar duruyor`, old.files[code].version === SCHEMA && MOBILE_TABLES.every(t => old.files[code].tables.includes(t)), { version: old.files[code].version });
    const d2 = stableDiff(goc.companies[code], back.companies[code]);
    check(`${code} ${OLD}'dan sonra yeniden aday: gösterdiği = göç ölçüsü (${d2.map(x => x.section).join(",") || "fark yok"})`, !d2.length, { d2: JSON.stringify(d2).slice(0, 1500) });

  writeJson("donus", { checks });

  summary(checks.length, checks.filter(c => !c.ok).length, `geri dönüş (aday → ${OLD} → aday)`);

} else if (step === "zincir") {
  const { stableDiff } = await imp("test/guvenilirlik/m30-tatbikat.mjs");
  const { diffFacts, oldDataMap } = await imp("test/guvenilirlik/m30-fark.mjs");
  const { scoped } = await imp("test/guvenilirlik/m30-uretici.mjs");
  const { fileFacts } = await imp("test/guvenilirlik/m30-olgular.mjs");

  const stepDir = path.join(root, `${name}-adim`);

  for (const d of [stepDir, directDir]) { rmSync(d, { recursive: true, force: true }); cpSync(source, d, { recursive: true }); }
  const ver = dir => dbFiles(dir).map(f => `${f.id}: ${readDb(f.file, db => db.prepare("PRAGMA user_version").get().user_version)}`).join(", ");
  log(`kaynak ${path.basename(source)}: şema ${ver(stepDir)}`);
  for (const version of pathVersions) {
    const started = performance.now();
    resetEdmCaches();
    const server = await boot(version, stepDir, day);
    try {
      const api = await server.login();
      for (const company of (await api.get("/api/companies")).data.all) await must(`${version} ${company.code} açılış`, scoped(api, company.id).get("/api/workspace/ledger/lock"));
    } finally {
      await server.close();
    }
    log(`  ${version} kendi koduyla açtı: şema ${ver(stepDir)} (${Math.round(performance.now() - started)} ms)`);
  }
  const last = pathVersions.at(-1);
  const lastCopy = path.join(root, `${name}-olcu`);

  cpSync(stepDir, lastCopy, { recursive: true });
