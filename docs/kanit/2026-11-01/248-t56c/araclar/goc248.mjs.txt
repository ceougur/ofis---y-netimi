async function measure({ version, dir, day }) {

    await edm.close();
    throw error;
  }
  const result = { version: version === CURRENT ? "aday" : version, commit: server.commit, day, bootMs: Math.round(performance.now() - started), firstAccessMs: {}, companies: {}, files: {} };
  try {
    const api = await server.login();
    result.install = await installFacts(api);
    for (const company of result.install.companies) {
      const t = performance.now();
      const s = scoped(api, company.id);

      const diffN = ar.filter((x, i) => JSON.stringify(x) !== JSON.stringify(strip([br[i] || {}])[0])).length;
      check(`${key}: wa.me hedefleri ${OLD} ↔ aday BİREBİR (${ar.length} cari; geçerli ${ar.filter(x => x.valid).length}/${br.filter(x => x.valid).length}; farklı satır ${diffN}; optOut≠false ${optOutBad})`, same && optOutBad === 0, { diffN, sample: ar.find((x, i) => JSON.stringify(x) !== JSON.stringify(strip([br[i] || {}])[0]))?.id });
    }
  }
  // (b) sayaç
  for (const c of old.companies) check(`${c.code} e-Belge sayacı (Bu Ay / 24 ay penceresi, iz) ${OLD} ↔ aday aynı (pencere satırı ${(old.usage[c.code].window || []).length})`, JSON.stringify(old.usage[c.code].window) === JSON.stringify(neu.usage[c.code].window) && JSON.stringify(old.usage[c.code].journal) === JSON.stringify(neu.usage[c.code].journal), { a: old.usage[c.code].window, b: neu.usage[c.code].window });
  // (c) geçmiş
  for (const key of Object.keys(old.history)) {
    const a = (old.history[key] || []).map(x => ({ id: x.id, kind: x.kind, status: x.status, body: x.body, at: x.at }));
    const b = (neu.history[key] || []).map(x => ({ id: x.id, kind: x.kind, status: x.status, body: x.body, at: x.at }));
    const sameAll = a.every(x => b.some(y => JSON.stringify(y) === JSON.stringify(x))) && a.length === b.length;
    if (a.length || b.length) check(`${key}: wa.me gönderim geçmişi aday kart geçmişinde aynı (${a.length} → ${b.length})`, sameAll, { a, b: neu.history[key] });
  }
  // (d) telefon
  const ph = neu.hook.phone;
  check(`${OLD}'de eşlenen telefon göçten sonra eşli (${ph.devices} cihaz), oturum ${ph.session}`, ph.devices === 1 && ph.session === 200, ph);
  for (const c of old.companies) check(`${c.code} göçten sonra telefondaki Kasa = masaüstü (${ph[c.code].status}) (hareket listesi hariç — 2.4.7 kararı "mobil Kasa yalnız bakiye/toplamlar"; ${OLD}'de de aynı: ${ph[c.code].oldSame}) ve = göçten önce telefonun gördüğü`, ph[c.code].status === 200 && ph[c.code].same && ph[c.code].oldSame && ph[c.code].farBefore, ph[c.code]);
  // (e) önizleme — beklenen PLANDAN: telefonsuz → "Telefon yok"; eksik hane → elenir; pasif → "Pasif Cari"; sabit hat, yurt dışı (+49, 0044), ülke kodsuz cep,
  // çok numaralı (ilki cep) → gönderilir (K1 A); aynı telefonlu iki cari → ikisi de gönderilir (K2 A); "İstemiyor" yok.
  for (const c of old.companies) {
    const p = neu.hook.preview[c.code];
    check(`${c.code} Elle Bağla ${p.conn}, onay ${p.ack}, Ekstre önizlemesi ${p.status}`, p.conn === 200 && p.ack === 200 && p.status === 200, { p: JSON.stringify(p).slice(0, 600) });
    const people = Object.fromEntries((p.data?.people || []).map(x => [x.id, x]));
    for (const acc of manifest.wa[c.code]) {
      const spec = WA_SPECS.find(x => x.key === acc.key);
      const got = people[acc.id];
      check(`${c.code} ${spec.name}: telefon "${spec.phone}" → beklenen ${spec.expect === "send" ? `gönderilir ${spec.e164}` : `elenir "${spec.expect}"`}; program: ${got ? (got.skip ? `elenir "${got.skip}"` : `gönderilir ${got.phone}`) : "listede yok"}`, got && (spec.expect === "send" ? !got.skip && got.phone === spec.e164 : got.skip === spec.expect), got);
    }
    // Bütün cariler: wa.me geçerli ⇒ API'de gönderilir; wa.me "Sabit hat" ⇒ API'de gönderilir (K1 A); öbür wa.me geçersizler API'de de elenir (aynı numara kuralı).
    const wame = Object.fromEntries((old.targets[`${c.code}/thisMonth`].recipients || []).map(x => [x.id, x]));
    const bad = [];
    for (const person of p.data?.people || []) {
      const w = wame[person.id];
      if (!w) { bad.push({ id: person.id, why: "wa.me listesinde yok" }); continue; }
      const passive = w && person.skip === "Pasif Cari";
      const expectSend = !passive && (w.valid || /^Sabit hat/.test(w.reason));
      if (!passive && expectSend !== !person.skip) bad.push({ id: person.id, w: [w.valid, w.reason, w.phone], api: person.skip });
    }
    const skipped = p.data?.skipped || {};
    check(`${c.code} önizleme bütün cariler (${p.data?.total}; gönderilebilir ${p.data?.sendable}; elenen ${JSON.stringify(skipped)}): wa.me kuralıyla tutarlı (uyuşmayan ${bad.length}); "WhatsApp Mesajı İstemiyor" 0; "Daha Önce Gönderildi" 0; "Zaten Kuyrukta" 0`, bad.length === 0 && !skipped["WhatsApp Mesajı İstemiyor"] && !skipped["Daha Önce Gönderildi"] && !skipped["Zaten Kuyrukta"] && p.data?.total === Object.keys(wame).length, { bad: bad.slice(0, 10) });
    log(`  ${c.code} Ödeme Hatırlatma önizlemesi ${p.reminder.status}: gönderilebilir ${p.reminder.sendable}, elenen ${JSON.stringify(p.reminder.skipped)}`);
  }
  // (f) veri dosyası
  for (const fl of dbFiles(dir)) {
    const info = readDb(fl.file, db => ({ v: db.prepare("PRAGMA user_version").get().user_version, n: db.prepare("SELECT count(*) AS n FROM accounts").get().n, opt: db.prepare("SELECT count(*) AS n FROM accounts WHERE wa_opt_out <> 0").get().n, jobs: db.prepare("SELECT count(*) AS n FROM wa_jobs").get().n, b: db.prepare("SELECT count(*) AS n FROM wa_batches").get().n, sends: db.prepare("SELECT count(*) AS n FROM message_sends").get().n }));
    check(`${fl.id} veri dosyası: şema ${info.v}; ${info.n} cari, "İstemiyor" ${info.opt}; wa_batches ${info.b}, wa_jobs ${info.jobs}; message_sends ${info.sends}`, info.v === SCHEMA && info.opt === 0 && info.b === 0 && info.jobs === 0 && info.sends === 3, info);
  }
  writeJson("wa", { checks, allDiff: neu.allDiff, preview: Object.fromEntries(Object.entries(neu.hook.preview).map(([k, v]) => [k, { total: v.data?.total, sendable: v.data?.sendable, skipped: v.data?.skipped, limit: v.data?.limit, reminder: v.reminder }])) });
  writeJson("wa-hedefler", { old: old.targets, neu: neu.targets });
  summary(checks.length, checks.filter(x => !x.ok).length, "WhatsApp eski veride: wa.me hedefleri, sayaç, geçmiş, eşli telefon, önizleme, wa_opt_out");
  process.exitCode = checks.some(x => !x.ok) ? 1 : 0;
  // İleri tarihli hareket denetimi ayrı adımda (veri klasörü burada kalır: wa-ileri).
} else {
  throw new Error(`bilinmeyen adım ${step}`);
}
