import { existsSync, mkdirSync, readFileSync, writeFileSync, appendFileSync } from "node:fs";
import path from "node:path";

const [W, ...bolumler] = process.argv.slice(2);
const S = "/tmp/claude-0/-home-user/2a941a30-cea4-5380-a679-f28a541c21a1/scratchpad/tamkapi248-6e1a";
const L = "/tmp/destekofis-olcum.lock";
const GUNLER = ["2026-11-01", "2026-11-02", "2026-11-03"].map(g => path.join(W, "docs/kanit", g, "248-tamkapi"));
const PW = "/tmp/claude-0/-home-user/2a941a30-cea4-5380-a679-f28a541c21a1/scratchpad/w1-248-q7m3/pw";

const now = () => new Date().toISOString().slice(11, 19);
const log = msg => { console.log(msg); appendFileSync(path.join(S, "kapi.log"), msg + "\n"); };
const hukum = ad => {
  for (const D of GUNLER) {
    const f = path.join(D, ad + ".json");
    if (!existsSync(f)) continue;
    try { return JSON.parse(readFileSync(f, "utf8")).hukum; } catch { return "OKUNAMADI"; }
  }
  return null;
};
const diskMB = () => {
  const r = spawnSync("df", ["--output=avail", "-BM", "/"], { encoding: "utf8" });
  return Number((r.stdout.trim().split("\n").pop() || "").replace(/\D/g, ""));
};
let n = 0;
const sonuc = [];
function tek(ad, cmd) {
  const bos = diskMB();
  log(`${now()} disk ${bos} MB boş — başlıyor ${ad}`);
  if (!(bos >= 5000)) { log(`DUR: disk ${bos} MB (ders 24)`); process.exit(3); }
  const r = spawnSync("flock", [L, "node", "tools/kanit.mjs", "kos", "248-tamkapi/" + ad, "--", ...cmd], { cwd: W, env, encoding: "utf8", maxBuffer: 1 << 28 });
  const out = (r.stdout || "") + (r.stderr || "");
  writeFileSync(path.join(S, "son.out"), out);
  const son = out.trim().split("\n").slice(-1)[0] || "";
  log(`${now()} ${ad} cikis=${r.status} hüküm=${hukum(ad)} ${son.slice(0, 220)}`);
  return r.status === 0 && hukum(ad) === "GEÇTİ";
}
function kos(ad, cmd) {
  n++;
  if (hukum(ad) === "GEÇTİ") { log(`ATLA ${ad} (GEÇTİ)`); sonuc.push([ad, "GEÇTİ"]); return; }
  if (hukum(ad) == null && tek(ad, cmd)) { sonuc.push([ad, "GEÇTİ"]); return; }
  if (hukum(ad + "-tekrar") === "GEÇTİ") { sonuc.push([ad, "1.kırmızı, 2.yeşil"]); return; }
  if (hukum(ad + "-tekrar") == null && tek(ad + "-tekrar", cmd)) { sonuc.push([ad, "1.kırmızı, 2.yeşil"]); return; }
  sonuc.push([ad, "2× kırmızı"]);
}

  if (b === "cidisi") {

    kos("cidisi-ui-ux-217", ["node", "--disable-warning=ExperimentalWarning", "test/e2e/ui-ux-217.mjs"]);

    kos("cidisi-is-akisi-248", ["node", "--disable-warning=ExperimentalWarning", "test/e2e/senaryo-is-akisi-248.mjs"]);

  if (b === "kahin-dis") {
