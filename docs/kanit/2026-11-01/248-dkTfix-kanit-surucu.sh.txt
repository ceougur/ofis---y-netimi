#!/bin/bash
# İş T kanıt sürücüsü: mutasyonlar (HEAD kopyası, ders 23/25) — karar tablosu testi, iki TZ, her koşu tools/kanit.mjs kos ile.
K=/tmp/claude-0/-home-user/2a941a30-cea4-5380-a679-f28a541c21a1/scratchpad/t248-kos
W=/home/user/wt-t2fix-248
cd $W || exit 1
export HUKUK_LICENSE_URL=http://127.0.0.1:9
T=whatsapp-t-durum-karar-tablosu-248.test.mjs
kos() {
  local ad=$1 kok=$2; shift 2
  local files=(); for f in "$@"; do files+=("$kok/test/$f"); done
  for tz in UTC Europe/Istanbul; do
    local out=$K/kos-$ad-${tz//\//_}.out
    TZ=$tz node tools/kanit.mjs kos "$ad-${tz//\//_}" -- node --disable-warning=ExperimentalWarning --test --test-concurrency=1 --test-timeout=300000 "${files[@]}" > $out 2>&1
    echo "$ad $tz çıkış=$? $(grep '\[kanit\]' $out | tail -1 | sed -E 's/\[kanit\] [^:]*: //' | cut -c1-100)"
    grep -E "^    not ok|^  not ok" $out | cut -c1-120 | sed 's/^/      /'
  done
}
echo "== MUTASYONLAR"
kok=$K/mut
rm -rf -- "${kok:?}"; mkdir -p $kok && git -C $W archive HEAD | tar -x -C $kok && rm -rf -- "$kok/docs/kanit" && ln -s /home/user/ofis---y-netimi/node_modules $kok/node_modules
for m in $(python3 $K/mut.py --list); do
  rm -rf -- "$kok/server" && git -C $W archive HEAD server | tar -x -C $kok
  python3 $K/mut.py $kok $m || continue
  kos "248-dkTfix-mutasyon-$m" $kok $T
done
rm -rf -- "${kok:?}"
echo BITTI
