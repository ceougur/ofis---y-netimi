import { readFileSync } from "node:fs";
import { DatabaseSync } from "node:sqlite";
const src = readFileSync("/home/user/wt-perf-248/server/lib/whatsapp/defter.mjs", "utf8");
const m = /const SENT_BEFORE_SQL = \(outer, tail\) => `([\s\S]*?)`;/.exec(src);
const sql = (o, t) => m[1].replace("${outer}", o).replace("${tail}", t);
const EK = "(SELECT group_concat(k2.period_key, char(31)) FROM claim_keys k2 WHERE k2.job_id = c.job_id AND k2.attempt = c.attempt) AS extra_keys,";

const sinceOld = "2025-01-01T00:00:00.000Z"; // pencere geniş: geçmişin tamamı eşleşmeye aday
for (const [ad, s] of [["dar pencere", since], ["geniş pencere", sinceOld]]) {

  const o2 = db.prepare(varyant.replace("${outer}", "SELECT accountId, at, period_key AS ownKey, (SELECT group_concat(k2.period_key, char(31)) FROM claim_keys k2 WHERE k2.job_id = c.job_id AND k2.attempt = c.attempt) AS extraKeys").replace("${tail}", ""));
  const keys = JSON.stringify(["secim|all", "ay|2026-02", "aralik|2026-02-01|2026-02-28"]);
  for (const [k, st] of [["özgün", o1], ["varyant", o2]]) {
    for (const [kad, kk] of [["2000 kişi anahtarı", own], ["genel anahtarlar", keys]]) {
      const t = performance.now();
      const rows = st.all("sirket-001", "statement", kk, kk, s, "", s);
      console.log(ad, k, kad, Math.round(performance.now() - t), "ms", rows.length, "satır", rows.length ? JSON.stringify(rows[0]).slice(0, 120) : "");
    }
  }
}
