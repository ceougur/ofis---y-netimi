#!/bin/bash
# Ana oturum: Ç5-A TUR 3a (yalnız taşıma) — YENİDEN KOŞU: kalıcı test iki TZ + TABAN KIRMIZISI + mutasyon + taşınan yedi kuyruk mutantı + BÜYÜK VERİDE BAYT BAYT EŞDEĞERLİK (ders 7, 16, 23, 38, 47).
# Depo değişkeni VERİLMEZ (ders 54). Yeni = tur 3a dalının ucu, taban = c5a3a-taban; ikisi de kendi `git archive` kopyam. Eşdeğerlik dökümü ajanın dok.mjs betiğiyle (yalnız okunur), veri = ana-kabul/temiz-100k'nin iki TAZE kopyası.
# Çağrı: flock -s -n olcum.lock timeout 6000 bash ana-c5a3a/yk.sh
S=/tmp/claude-0/-home-user/de5e1cbf-7e68-5683-b7bb-f27655eaed4f/scratchpad
R=/home/user/ofis---y-netimi; K=$S/ana-c5a3a/kanit; Y=$S/ana-c5a3a/yeni; M=$S/ana-c5a3a/mut; TB=$S/ana-c5a3a/taban; A=worktree-agent-a068d42a7b77ba335; KAYNAK=$S/ana-kabul/temiz-100k
export HUKUK_LICENSE_URL=http://127.0.0.1:9
say() { grep -E "^# (tests|pass|fail|cancelled|skipped)" "$1" | tr '\n' ' '; }
echo "başladı $(date -u +%T) yük $(cut -d' ' -f1-3 /proc/loadavg)"
for c in "$Y $A" "$M $A" "$TB c5a3a-taban"; do set -- $c; if [ ! -d $1 ]; then mkdir -p $1 && git -C $R archive $2 ':!docs/kanit' | tar -x -C $1 && ln -s $R/node_modules $1/node_modules || exit 1; fi; done
T=test/obligation-input-250.test.mjs; H=test/fixtures/obligation-input-250-hakem.json
[ -f $TB/$T ] && { echo "taban kopyasında yeni test ZATEN var — beklenmedik"; exit 1; }
cd $Y || exit 1
for tz in UTC Europe/Istanbul; do m=${tz//\//-}
  TZ=$tz timeout 1500 node --disable-warning=ExperimentalWarning --test $T > $K/yeni-$m.out 2>&1; echo "YENİ girdi modülü testi $tz çıkış $? $(say $K/yeni-$m.out)"
done
cp $Y/$T $TB/$T && cp $Y/$H $TB/$H || exit 1
(cd $TB && TZ=UTC timeout 900 node --disable-warning=ExperimentalWarning --test $T > $K/taban-test.out 2>&1; echo "TABAN girdi modülü testi çıkış $? $(say $K/taban-test.out)")
grep -m2 -E "yüklenemedi|Cannot find module" $K/taban-test.out | cut -c1-200
rm $TB/$T $TB/$H
timeout 2400 node test/obligation-input-250-mutasyon.mjs --kopya $M > $K/mutasyon.out 2>&1; echo "MUTASYON çıkış $? :: $(tail -2 $K/mutasyon.out | tr '\n' ' ' | cut -c1-260)"
mkdir -p $S/ana-c5a3a/tyk-mut && timeout 2400 node test/tyk-mutasyon-250.mjs --klasor $S/ana-c5a3a/tyk-mut --yalniz O1-API,O1C-a,O1C-b,O1C-d,D05,D06,KD5-c > $K/tyk-mutasyon.out 2>&1; echo "TAŞINAN yedi kuyruk mutantı çıkış $? :: $(tail -3 $K/tyk-mutasyon.out | tr '\n' ' ' | cut -c1-300)"
for KOD in taban yeni; do
  ROOT=$TB; [ $KOD = yeni ] && ROOT=$Y
  KOPYA=/dev/shm/ana-c5a3a-esd-$KOD
  [ -e "$KOPYA" ] && { echo "kopya zaten var: $KOPYA"; exit 1; }
  cp -a "$KAYNAK" "$KOPYA" || exit 1
  s=$(date +%s)
  timeout -k 5 3000 node --disable-warning=ExperimentalWarning $S/c5a3a-t1/betik/dok.mjs "$ROOT" "$KOPYA" $K/esd-$KOD.jsonl "2026-10-08T12:00:00+03:00" > $K/esd-$KOD.ozet 2>&1
  echo "eşdeğerlik dökümü $KOD çıkış $? · $(( $(date +%s) - s )) sn · yük $(cut -d' ' -f1 /proc/loadavg) · $(tail -1 $K/esd-$KOD.ozet | cut -c1-160)"
  rm -rf /dev/shm/ana-c5a3a-esd-$KOD
done
if cmp -s $K/esd-taban.jsonl $K/esd-yeni.jsonl; then echo "SONUÇ: BAYT BAYT AYNI ($(wc -l < $K/esd-yeni.jsonl) satır; sha256 $(sha256sum < $K/esd-yeni.jsonl | cut -c1-16); boyut $(du -h $K/esd-yeni.jsonl | cut -f1))"; else echo "SONUÇ: FARK VAR"; cmp $K/esd-taban.jsonl $K/esd-yeni.jsonl | head -2; fi
sha256sum $K/esd-taban.jsonl $K/esd-yeni.jsonl | cut -c1-80 > $K/esd-sha256.txt; rm -f $K/esd-taban.jsonl $K/esd-yeni.jsonl
echo "bitti $(date -u +%T) yük $(cut -d' ' -f1-3 /proc/loadavg) · disk $(df -h / | awk 'NR==2{print $4}')"
