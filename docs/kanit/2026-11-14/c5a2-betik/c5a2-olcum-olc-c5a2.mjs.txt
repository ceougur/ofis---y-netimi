// Kullanım: node olc-c5a2.mjs <kod kökü> <veri kopyası> [tekrar=5]   (özel ölçüm kilidiyle çağrılır; kilitli kipte kopya önceden kilit.py ile hazırlanır)
import { readFileSync } from "node:fs";
import path from "node:path";
import { monitorEventLoopDelay } from "node:perf_hooks";
import { pathToFileURL } from "node:url";

process.env.HUKUK_LICENSE_URL ||= "http://127.0.0.1:9";
const [root, dir, repeatText = "5"] = process.argv.slice(2);
const repeat = Number(repeatText) || 5;
const load = () => readFileSync("/proc/loadavg", "utf8").trim().split(" ").slice(0, 3).join(" ");
const { createApp } = await import(pathToFileURL(path.join(root, "server", "app.mjs")).href);
const { createClient } = await import(pathToFileURL(path.join(root, "test", "helpers.mjs")).href);
const PASS = "Kapi-Olcum-2026!";
const t0 = performance.now();
const app = createApp({
  dataDir: path.join(dir, "data"), backupDir: path.join(dir, "backups"), logLevel: "silent", scheduleBackups: false,
  env: { HUKUK_ADMIN_PASSWORD: PASS, HUKUK_DATASET_AUTOSYNC: "0" }, license: { enforce: false, machineId: "0123456789abcdef0123456789abcdef" },
  startLicenseTimers: false, integrityScan: false, analysisWorker: false, now: "2026-10-08T12:00:00+03:00", moneyStrict: false,
});
const { port } = await app.listen(0, "127.0.0.1");
const r = v => Math.round(v * 10) / 10;
console.log(`kod ${root} · yük (başta) ${load()} · TZ ${process.env.TZ || "(yerel)"}`);
console.log(`  AÇILIŞ (göç dahil): durum 200 · en iyi ${r(performance.now() - t0)} ms · ortanca ${r(performance.now() - t0)} ms · olay döngüsü en çok 0 ms (n=1)`);
{
  const st = app.ledger.snapshots.state();
  if (st.pending) {
    const d0 = performance.now();
    await app.ledger.snapshots.runPending();
    const ms = performance.now() - d0;
    console.log(`  DONMA (bekleyen koşu): durum 200 · en iyi ${r(ms)} ms · ortanca ${r(ms)} ms · olay döngüsü en çok 0 ms (n=1)`);
    console.log(`donma kapsamı ${app.ledger.snapshots.state().coverage} (kural ${st.pending.rule}, ${st.pending.through})`);
  } else console.log(`donma: bekleyen koşu yok (kapsam ${st.coverage || "—"})`);
}
const client = createClient(`http://127.0.0.1:${port}`);
if ((await client.login("admin", PASS)).status !== 200) throw new Error("giriş");
const store = app.store;
const acc = store.get("SELECT a.id FROM accounts a WHERE a.deleted_at IS NULL AND a.type = 'customer' ORDER BY (SELECT COUNT(*) FROM account_entries e WHERE e.account_id = a.id) DESC LIMIT 1").id;
const item = store.get("SELECT l.item_id AS id, l.unit_price AS price, l.vat_rate AS vat FROM invoice_lines l JOIN invoices i ON i.id = l.invoice_id WHERE i.scenario = 'service_sale' AND l.item_id <> '' ORDER BY i.issue_date DESC LIMIT 1")
  || store.get("SELECT l.item_id AS id, l.unit_price AS price, l.vat_rate AS vat FROM invoice_lines l JOIN invoices i ON i.id = l.invoice_id WHERE i.scenario = 'goods_sale' AND l.item_id <> '' ORDER BY i.issue_date DESC LIMIT 1");
const h = monitorEventLoopDelay({ resolution: 5 });
h.enable();
const results = new Map();
const govde = new Map();
const timed = async (label, call) => {
  h.reset();
  const s = performance.now();
  const res = await call();
  const ms = performance.now() - s;
  if (!results.has(label)) results.set(label, { ms: [], loop: [], status: new Set() });
  const row = results.get(label);
  row.ms.push(ms); row.loop.push(h.max / 1e6); row.status.add(res.status);
  if (res.status !== 200 && !govde.has(label)) govde.set(label, JSON.stringify(res.data ?? res.body ?? "").slice(0, 200));
  return res;
};
const RANGE = "from=2026-01-01&to=2026-12-31";
for (let i = 0; i < repeat; i += 1) {
  await timed("Mizan + mutabakat (GET ledger)", () => client.get("/api/workspace/ledger"));
  await timed("Yevmiye Defteri (rapor)", () => client.get(`/api/workspace/report-center/yevmiye?${RANGE}`));
  await timed("Hesap Planı Mizanı (rapor)", () => client.get(`/api/workspace/report-center/hesap-mizani?${RANGE}`));
  await timed("Alt Hesap Mizanı (rapor)", () => client.get(`/api/workspace/report-center/alt-hesap-mizani?${RANGE}`));
  await timed("Defter Mutabakatı (rapor)", () => client.get("/api/workspace/report-center/defter-mutabakati"));
  await timed("Mutabakat Testi (GET ledger/integrity)", () => client.get("/api/workspace/ledger/integrity"));
  await timed("Yevmiye Defteri bütün zaman (rapor)", () => client.get("/api/workspace/report-center/yevmiye?from=2000-01-01&to=2099-12-31"));
  await timed("YAZIM Borç Yaz", () => client.post(`/api/workspace/accounts/${acc}/entries`, { kind: "debt", amount: "10", note: `ölçüm ${i}`, date: "2026-10-08" }));
  await timed("YAZIM evrak (alınan çek, carili)", () => client.post("/api/workspace/cheques", { direction: "in", instrument: "cheque", serialNo: `OLC-${i}-${Date.now()}`, bank: "Akbank", accountId: acc, amount: "10", issueDate: "2026-10-08", dueDate: "2026-12-08" }));
  const inv = await timed("YAZIM fatura (satış, açık hesap)", () => client.post("/api/workspace/invoices", { scenario: "service_sale", accountId: acc, issueDate: "2026-10-08", lines: [{ itemId: item.id, qty: 1, unitPrice: 100, vatRate: Number(item.vat) || 0 }], payment: { rest: "open" } }));
  const invId = inv.data?.data?.id || inv.data?.id;
  if (i === 0) console.log(`fatura yanıtı anahtarları: ${Object.keys(inv.data || {}).slice(0, 12).join(",")} · id ${invId}`);
  if (invId) await timed("YAZIM fatura iptali", () => client.post(`/api/workspace/invoices/${invId}/cancel`, { reason: "ölçüm", cashForce: true, negativeOk: true }));
  await timed("cari kartı", () => client.get(`/api/workspace/accounts/${acc}`));
  await timed("Açılış Olabilir listesi (sayfasız)", () => client.get("/api/workspace/opening-review"));
  await timed("Açılış Olabilir listesi sayfalı (yeni; tabanda limit yok sayılır)", () => client.get("/api/workspace/opening-review?limit=200&offset=0&all=1"));
  await timed("Zil sayısı (yeni; tabanda tam liste)", () => client.get("/api/workspace/opening-review?counts=1"));
  await timed("Kural Farkı Düzeltme ön izlemesi (yeni; tabanda 404)", () => client.get("/api/admin/ledger/rule-diff-correction"));
  await timed("Açılış Eşleştirme GET (yeni; tabanda 404)", () => client.get("/api/admin/ledger/opening-match?date=2026-10-08"));
  await timed("Açılış Eşleştirme ön izleme (yeni; tabanda 404)", () => client.post("/api/admin/ledger/opening-match/preview", { date: "2026-10-08", lines: [{ account: "500", side: "credit", amount: "1" }] }));
}
const median = list => [...list].sort((a, b) => a - b)[Math.floor(list.length / 2)];
for (const [label, row] of results) console.log(`  ${label}: durum ${[...row.status].join("/")} · en iyi ${r(Math.min(...row.ms))} ms · ortanca ${r(median(row.ms))} ms · olay döngüsü en çok ${r(Math.max(...row.loop))} ms (n=${row.ms.length})`);
for (const [label, body] of govde) console.log(`gövde (200 değil) ${label}: ${body}`);
console.log(`yük (sonda) ${load()}`);
await app.close();
