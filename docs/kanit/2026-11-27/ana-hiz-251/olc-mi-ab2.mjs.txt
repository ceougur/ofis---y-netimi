import { readFileSync } from "node:fs";
import path from "node:path";
import { monitorEventLoopDelay } from "node:perf_hooks";
import { pathToFileURL } from "node:url";

process.env.HUKUK_LICENSE_URL ||= "http://127.0.0.1:9";
const [baseRoot, baseDir, newRoot, newDir, pairsText = "10"] = process.argv.slice(2);
const pairs = Number(pairsText) || 10;
const load = () => readFileSync("/proc/loadavg", "utf8").trim().split(" ").slice(0, 3).join(" ");
console.log(`taban ${baseRoot} · yeni ${newRoot} · yük (başta) ${load()}`);
const PASS = "Kapi-Olcum-2026!";
async function boot(root, dir) {
  const { createApp } = await import(pathToFileURL(path.join(root, "server", "app.mjs")).href);
  const { createClient } = await import(pathToFileURL(path.join(root, "test", "helpers.mjs")).href);
  const app = createApp({
    dataDir: path.join(dir, "data"), backupDir: path.join(dir, "backups"), logLevel: "silent", scheduleBackups: false,
    env: { HUKUK_ADMIN_PASSWORD: PASS, HUKUK_DATASET_AUTOSYNC: "0" }, license: { enforce: false, machineId: "0123456789abcdef0123456789abcdef" },
    startLicenseTimers: false, integrityScan: false, analysisWorker: false, now: "2026-10-08T12:00:00+03:00", moneyStrict: false,
  });
  const { port } = await app.listen(0, "127.0.0.1");
  const client = createClient(`http://127.0.0.1:${port}`);
  if ((await client.login("admin", PASS)).status !== 200) throw new Error("giriş");
  return { app, client };
}
const A = await boot(baseRoot, baseDir);
const B = await boot(newRoot, newDir);
const h = monitorEventLoopDelay({ resolution: 1 });
h.enable();
const r = v => Math.round(v * 10) / 10;
const PATHS = [
  ["Liste dönemsiz (Aktif, 300)", "/api/workspace/accounts?status=active&limit=300"],
  ["Liste dönemsiz (Tümü, 300)", "/api/workspace/accounts?status=all&limit=300"],
];
const out = new Map();
const timed = async (side, label, client, url) => {
  h.reset();
  const s = performance.now();
  const res = await client.get(url);
  const ms = performance.now() - s;
  const key = `${label} · ${side}`;
  if (!out.has(key)) out.set(key, []);
  out.get(key).push({ ms, loop: h.max / 1e6, status: res.status, same: JSON.stringify(res.data) });
};
for (const [, url] of PATHS) {
  await A.client.get(url);
  await B.client.get(url);
}
for (let i = 0; i < pairs; i += 1) {
  const order = i % 2 ? [["yeni", B], ["taban", A]] : [["taban", A], ["yeni", B]];
  for (const [label, url] of PATHS) for (const [side, sys] of order) await timed(side, label, sys.client, url);
}
const sorted = list => [...list].sort((a, b) => a - b);
const q = (list, p) => sorted(list)[Math.min(list.length - 1, Math.floor(list.length * p))];
for (const [label] of PATHS) {
  const a = out.get(`${label} · taban`).map(x => x.ms);
  const b = out.get(`${label} · yeni`).map(x => x.ms);
  const st = l => `en az ${r(Math.min(...l))} · ç1 ${r(q(l, 0.25))} · orta ${r(q(l, 0.5))} · ç3 ${r(q(l, 0.75))} · en çok ${r(Math.max(...l))}`;
  let yeniYavas = 0;
  for (let k = 0; k < a.length; k += 1) if (b[k] > a[k]) yeniYavas += 1;
  const pct = (x, y) => r((y / x - 1) * 100);
  console.log(`  ${label} · ${a.length} çift`);
  console.log(`    taban: ${st(a)}`);
  console.log(`    yeni : ${st(b)}`);
  console.log(`    fark: en az ${pct(Math.min(...a), Math.min(...b))} % · ç1 ${pct(q(a, 0.25), q(b, 0.25))} % · orta ${pct(q(a, 0.5), q(b, 0.5))} % · yeninin yavaş olduğu çift ${yeniYavas}/${a.length}`);
}
console.log(`yük (sonda) ${load()}`);
await A.app.close();
await B.app.close();
