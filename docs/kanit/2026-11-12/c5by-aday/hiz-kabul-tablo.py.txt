for p in sorted(glob.glob(S+'/*.txt')):
    ad=os.path.basename(p).split('-')[0]
    for l in open(p,encoding='utf-8',errors='replace'):
        m=re.match(r'açılış (\d+) ms',l)
        if m: k='açılış'; T.setdefault((k,ad),[]).append(float(m.group(1))); (k in order) or order.append(k); continue

        if m: yuk.setdefault(ad,[]).append(float(m.group(1))); continue
        m=re.match(r'\s*(.+?)(?: #\d)?(?: \((?:cari|en kalabalık).*?\))?: (?:durum )?(\d+) · (?:toplam )?([\d.]+) ms',l)
        if m:
            k=m.group(1).strip()[:34]
            if m.group(2)!='200': k+=' [durum '+m.group(2)+']'
            T.setdefault((k,ad),[]).append(float(m.group(3))); (k in order) or order.append(k)
print('yükler:',{a:(min(v),max(v)) for a,v in yuk.items()})

    print(f"{k:36} {A[0]:.0f}/{A[1]:.0f} ({A[2]}) | {B[0]:.0f}/{B[1]:.0f} ({B[2]}) | {f(B[0],A[0]):+.0f}/{f(B[1],A[1]):+.0f}{i}")
