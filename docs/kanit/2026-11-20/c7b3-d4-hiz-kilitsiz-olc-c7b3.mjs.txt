});
const { port } = await app.listen(0, "127.0.0.1");
const r = v => Math.round(v * 10) / 10;
console.log(`kod ${root} · yük (başta) ${load()} · TZ ${process.env.TZ || "(yerel)"}`);
if (app.ledger.snapshots.state().pending) await app.ledger.snapshots.runPending();

const coverAcc = store.get("SELECT p.account_id AS id FROM plans p WHERE p.deleted_at IS NULL AND p.status = 'active' AND p.covers_balance = 1 AND p.invoice_id = '' AND p.account_id <> '' ORDER BY p.rowid DESC LIMIT 1")?.id || acc;

console.log(`cari ${acc} · kapsayan kartlı cari ${coverAcc} · stok kalemi ${stockItem || "YOK"}`);
const h = monitorEventLoopDelay({ resolution: 5 });
h.enable();
const results = new Map();
const govde = new Map();
const timed = async (label, call) => {
  h.reset();
  const s = performance.now();
  const res = await call();
  const ms = performance.now() - s;
  if (!results.has(label)) results.set(label, { ms: [], loop: [], status: new Set() });
  const row = results.get(label);
  row.ms.push(ms); row.loop.push(h.max / 1e6); row.status.add(res.status);
  if (res.status !== 200 && !govde.has(label)) govde.set(label, JSON.stringify(res.data ?? res.body ?? "").slice(0, 200));
  return res;
};
const idOf = res => res.data?.data?.id || res.data?.id;
const inst = { rest: "installments", installments: { count: 3, firstDue: "2026-10-08" } };

  if (tId) await timed("Fatura kartı (GET invoices/:id, İptal Et notu)", () => client.get(`/api/workspace/invoices/${tId}`));
  if (planId) await timed("YAZIM taksit tahsilatı", () => client.post(`/api/workspace/plans/${planId}/entries`, { kind: "in", amount: "300", date: "2026-10-08", method: "cash" }));

  }
  const u = await timed("YAZIM taksitli satış (tahsilatsız)", () => client.post("/api/workspace/invoices", { scenario: "service_sale", accountId: acc, issueDate: "2026-10-08", lines: [{ name: "Ölçüm Hizmeti", qty: 1, unitPrice: 500, vatRate: 20 }], payment: inst }));

}
