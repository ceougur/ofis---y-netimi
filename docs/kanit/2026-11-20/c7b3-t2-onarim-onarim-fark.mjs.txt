// Kullanım: node onarim-fark.mjs <kod kökü>
import { readFileSync } from "node:fs";
import path from "node:path";
import { pathToFileURL } from "node:url";
const root = process.argv[2];
process.env.HUKUK_LICENSE_URL = "http://127.0.0.1:9";
const imp = p => import(pathToFileURL(path.join(root, p)).href);
const { readFixture, unpackFixture } = await imp("test/guvenilirlik/fikstur.mjs");
const { ONARIM_FIXTURE } = await imp("test/guvenilirlik/onarim-hiz-veri.mjs");
const { openWith } = await imp("test/guvenilirlik/onarim-dokum.mjs");
const fixture = readFixture(ONARIM_FIXTURE);
const expected = JSON.parse(readFileSync(path.join(root, "test", "fixtures", "onarim-hiz-250-beklenen.json"), "utf8")).dokum;
const unpacked = unpackFixture(ONARIM_FIXTURE);
const { app } = await openWith(root, { dataDir: unpacked.dataDir, backupDir: unpacked.backupDir, today: fixture.today });
const store = app.store;
const second = app.context.invoices.repairOwnCards({});
const lock = app.context.period?.lockedUntil?.() || store.getSetting?.("period.lockedUntil") || "?";
console.log(`bugün ${fixture.today} · kilit ${lock}`);
const exp = expected.repair || {};
const key = r => `${r.number}:${r.from}->${r.to}`;
const ef = new Set((exp.fixed || []).map(key)), el = new Set((exp.locked || []).map(r => r.number));
console.log(`eski döküm: onarılan ${(exp.fixed || []).length}, kilitli ${(exp.locked || []).length} · yeni ikinci koşu kilitli ${second.locked.length}`);
for (const r of second.locked) {
  if (el.has(r.number)) continue;
  const items = store.all("SELECT due_date AS dueDate, amount FROM plan_items WHERE plan_id = ? ORDER BY due_date, seq", r.planId);
  const wasFixed = (exp.fixed || []).find(f => f.number === r.number);
  console.log(`YENİ KİLİTLİ ${r.number} kart ${r.planId} kalan ${r.left} hedef ${r.target} · eski dökümde ${wasFixed ? `ONARILMIŞTI ${wasFixed.from} → ${wasFixed.to}` : "yoktu"} · taksitler ${items.map(i => `${i.dueDate}:${i.amount}`).join(" ")}`);
}
await app.close();
unpacked.cleanup();
