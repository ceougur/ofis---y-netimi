for k in yeni:/home/user/ofis---y-netimi once:$S/kahin8y-ag c5aonce:$A/c5aonce; do
  ad=${k%%:*}; R=${k#*:}
  for c in ${CASES:-late-covers}; do for ro in 0 1; do
    timeout -k 5 300 env GIT_DIR=/home/user/ofis---y-netimi/.git MODE=case TPL=$N/tpl WORK=$A/work ROOT=$R CASE=$c REOPEN=$ro TZ=${TZ:-Europe/Istanbul} HUKUK_LICENSE_URL=http://127.0.0.1:9 node --disable-warning=ExperimentalWarning probe.mjs > $A/$ad-$c-$ro.json 2> $A/$ad-$c-$ro.err; echo "$ad $c reopen=$ro çıkış $? · $(head -c 900 $A/$ad-$c-$ro.json)"
  done; done
done
