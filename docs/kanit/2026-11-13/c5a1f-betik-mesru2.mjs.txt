// Ç5-A tur 1f — İş 0 (b) kanıtı: kilitli tarihli satırlarda kapıdan geçen MEŞRU yollar — yanıt kodu, sessiz mi (bulgu / günlük / zil),
// kilitli dönemin canlı türetmesi değişti mi. ROOT=<kod> TPL=<şablon> node mesru2.mjs
import { cpSync, mkdtempSync, readFileSync, rmSync } from "node:fs";
import path from "node:path";
import { pathToFileURL } from "node:url";
process.env.HUKUK_LICENSE_URL = "http://127.0.0.1:9";
const ROOT = process.env.ROOT;
const TPL = process.env.TPL;
const imp = rel => import(pathToFileURL(path.join(ROOT, rel)).href);
const { CURRENT, bootVersion } = await imp("test/guvenilirlik/surumler.mjs");
const { ids, LOCK } = JSON.parse(readFileSync(path.join(TPL, "ids.json"), "utf8"));
const NOW = { time: "2026-10-20T12:00:00+03:00", fixed: true };
const quiet = { debug() {}, info() {}, warn() {}, error() {} };
const root = mkdtempSync(path.join(path.dirname(TPL), "c5a1f-mesru-"));
cpSync(TPL, root, { recursive: true });
const dirs = { dataDir: path.join(root, "data"), backupDir: path.join(root, "backups") };
const KNOWN = new Set(["ledger:rule-diff"]);
const codes = list => (list || []).map(item => item.code);
const lines = (entries, lock) => {
  const out = new Map();
  for (const entry of entries) {
    if (!(String(entry.date) <= lock)) continue;
    for (const line of entry.lines || []) {
      const key = `${entry.date}|${line.account}|${line.sub || ""}|${line.party || ""}|${line.debit || 0}|${line.credit || 0}`;
      out.set(key, (out.get(key) || 0) + 1);
    }
  }
  return out;
};
const diff = (a, b) => {
  let n = 0;
  const sample = [];
  for (const key of new Set([...a.keys(), ...b.keys()])) {
    const d = (b.get(key) || 0) - (a.get(key) || 0);
    if (d) { n += Math.abs(d); if (sample.length < 4) sample.push(`${d > 0 ? "+" : "-"}${key}`); }
  }
  return { n, sample };
};
const steps = [];
let server = await bootVersion(CURRENT, { dataDir: dirs.dataDir, backupDir: dirs.backupDir, now: NOW, integrityScan: false, log: quiet });
try {
  const app = server.app;
  await app.ledger.snapshots.runPending();
  const api = await server.login();
  const alerts = [];
  app.events.tap((event, payload) => { if (event === "integrity.alert") alerts.push(codes(payload.findings).join("+")); });
  const logN = () => app.store.get("SELECT COUNT(*) AS n FROM integrity_log").n;
  let lockNow = app.ledger.snapshots.state().lock;
  let base = lines(app.ledger.build(), lockNow);
  async function step(name, fn, { rebase = false } = {}) {
    const log0 = logN();
    const z0 = alerts.length;
    let res;
    try { res = await fn(); } catch (error) { res = { status: "HATA", data: String(error?.message || error).slice(0, 200) }; }
    await app.ledger.snapshots.runPending?.();
    const s = app.integrity.scan();
    const w = await app.integrity.scanInBackground();
    const msg = res && typeof res.data === "object" && res.data && !Array.isArray(res.data) ? (res.data.error || res.data.message || "") : "";
    const now = lines(app.ledger.build(), lockNow);
    steps.push({ step: name, status: res?.status, msg: String(msg).slice(0, 120), findings: [...new Set([...codes(s.findings), ...codes(w.findings)])], fails: codes(s.result?.failures).filter(c => !KNOWN.has(c)), log: logN() - log0, zil: alerts.slice(z0), derive: diff(base, now) });
    if (rebase) { lockNow = app.ledger.snapshots.state().lock; base = lines(app.ledger.build(), lockNow); }
    return res;
  }
  await step("carisiz evrak bugün tahsil (T1F-7)", () => api.post(`/api/workspace/cheques/${ids.cheque7}/actions`, { action: "collect", date: "2026-10-20", method: "cash" }));
  await step("carisiz evrak bugün karşılıksız, cari seçilmeden (T1F-8)", () => api.post(`/api/workspace/cheques/${ids.cheque8}/actions`, { action: "bounce", date: "2026-10-20" }));
  await step("carisiz evrak bugün karşılıksız + cari seçimi (T1F-3 → A)", () => api.post(`/api/workspace/cheques/${ids.cheque3}/actions`, { action: "bounce", date: "2026-10-20", accountId: ids.A }));
  await step("carili evrak bugün karşılıksız (T1F-9)", () => api.post(`/api/workspace/cheques/${ids.cheque9}/actions`, { action: "bounce", date: "2026-10-20" }));
  await step("carili evrak bugün ciro (T1F-1 → S)", () => api.post(`/api/workspace/cheques/${ids.cheque1}/actions`, { action: "endorse", date: "2026-10-20", accountId: ids.S }));
  await step("kilitte tahsile verilen evrak bugün tahsil (T1F-4; kayıt kapanır)", () => api.post(`/api/workspace/cheques/${ids.cheque4}/actions`, { action: "collect", date: "2026-10-20", method: "bank" }));
  await step("kilitli karşılıksızı geri al (T1F-2) — beklenen 409", () => api.post(`/api/workspace/cheques/${ids.cheque2}/undo`, {}));
  await step("kilitli evrakın carisini değiştir (T1F-7 düzenle) — beklenen 409", () => api.put(`/api/workspace/cheques/${ids.cheque9}`, { accountId: ids.S }));
  await step("kilitli taksit kartının Kayıt Tarihi boş — beklenen 400/409", () => api.put(`/api/workspace/plans/${ids.plan}`, { registeredOn: "" }));
  steps.push({ step: "Kayıt Tarihi (boş istekten sonra)", registeredOn: app.store.get("SELECT registered_on AS r FROM plans WHERE id = ?", ids.plan).r });
  await step("hareketli banka hesabının alt hesabı (glSub) — SINIF 3", () => api.put(`/api/workspace/bank/accounts/${ids.bank}`, { glSub: "102.99" }));
  steps.push({ step: "alt hesap (istekten sonra)", glSub: app.store.get("SELECT gl_sub AS g FROM bank_accounts WHERE id = ?", ids.bank).g });
  await step("kilitli taksit kartını kapat (bugün)", () => api.put(`/api/workspace/plans/${ids.plan}`, { status: "closed" }));
  await step("kilitli taksit kartını sil — beklenen 409", () => api.del(`/api/workspace/plans/${ids.importedPlan}`));
  await step("kilitli tahsilatın fatura bağı (düzenle) — beklenen 409", () => api.put(`/api/workspace/accounts/${ids.A}/entries/${ids.entryIn}`, { kind: "in", amount: "500", date: "2026-10-09", method: "cash", invoiceId: ids.invSale }));
  // Kilitli tahsilat ↔ kilit SONRASI fatura bağı: kilit kalkar, fatura 15.10, tahsilat 09.10 o faturaya bağlanır, kilit yeniden; fatura bugün iptal.
  await step("kilidi kaldır", () => api.put("/api/admin/period-lock", { lockedUntil: "" }), { rebase: true });
  const inv = await step("fatura 15.10 (A)", () => api.post("/api/workspace/invoices", { scenario: "service_sale", accountId: ids.A, issueDate: "2026-10-15", lines: [{ name: "Danışmanlık", qty: 1, unitPrice: 300, vatRate: 0 }], payment: { rest: "open" } }));
  await step("tahsilat 09.10 → 15.10 faturasına bağlı", () => api.post(`/api/workspace/accounts/${ids.A}/entries`, { kind: "in", amount: "100", date: "2026-10-09", method: "cash", invoiceId: inv?.data?.id || "" }));
  await step("yeniden kilitle 10.10", () => api.put("/api/admin/period-lock", { lockedUntil: LOCK }), { rebase: true });
  steps.push({ step: "bağ", linked: app.store.get("SELECT COUNT(*) AS n FROM account_entries WHERE invoice_id = ? AND date <= ?", inv?.data?.id || "-", LOCK).n });
  await step("15.10 faturasını bugün iptal et", () => api.post(`/api/workspace/invoices/${inv?.data?.id}/cancel`, { reason: "deneme" }));
  steps.push({ step: "bağ (iptalden sonra)", linked: app.store.get("SELECT COUNT(*) AS n FROM account_entries WHERE invoice_id = ? AND date <= ?", inv?.data?.id || "-", LOCK).n });
} catch (error) {
  steps.push({ error: String(error?.stack || error).slice(0, 1200) });
} finally {
  await server?.close().catch(() => {});
  rmSync(root, { recursive: true, force: true });
}
for (const s of steps) console.log(JSON.stringify(s));
