#!/bin/bash
# Ana oturum: Ç5-A TUR 1f ADAYI — İKİNCİ GEÇİŞ (kâhin paketi ana dala girdikten sonra): sınama ağacında çalışma dalı ucu YENİDEN alınır + tur 1f dalı;
# yeni testler (UTC; hızlı doğrulama), liste yeniden üretimi (fark BOŞ olmalı), ci.yml'deki BÜTÜN kâhin adımları, dört kapı testi, sonra TAM npm test.
# Hata durdurmaz (kurulum dışında); her adımın çıkışı yazılır. Çağrı: flock -s -n olcum.lock timeout 7000 bash c5a1fy-aday2.sh
S=/tmp/claude-0/-home-user/de5e1cbf-7e68-5683-b7bb-f27655eaed4f/scratchpad
W=$S/d2t3-ya-ag; K=$S/c5a1fy-kanit2; L=test/bagimsiz/cari/YAPIM-ASAMASI.json
AJAN=worktree-agent-af550e32f1aedd444
export KANIT_DIR=$K HUKUK_LICENSE_URL=http://127.0.0.1:9 PLAYWRIGHT_BROWSERS_PATH=$S/pw-1194
G="git -c user.name=ana -c user.email=ana@yerel"
YENI="test/kilit-izi-alanlar-250.test.mjs test/defter-kaynak-iz-250.test.mjs"
say() { grep -E "^# (tests|pass|fail|cancelled|skipped)" "$1" | tr '\n' ' '; }
mkdir -p $K || exit 1
cd $W || exit 1
[ -z "$(git status -s | grep -v node_modules)" ] || { echo "sınama ağacı temiz değil"; git status -s | head -5; exit 1; }
ESKI=$(git rev-parse --short HEAD)
git checkout -q -B sinama-c5a1f claude/epic-thompson-w0lc2l || exit 1
$G merge -q --no-ff --no-verify -m "SINAMA: + Ç5-A tur 1f (gönderilmez)" $AJAN > $K/merge.out 2>&1 || { echo "tur 1f birleşimi çakıştı"; git status -s | grep -E '^(UU|AA)' | head; exit 1; }
[ -e node_modules ] || ln -s /home/user/ofis---y-netimi/node_modules node_modules
echo "başladı $(date -u +%T) yük $(cut -d' ' -f1-3 /proc/loadavg) · aday $(git rev-parse --short HEAD) · önceki aday $ESKI ile ÜRÜN farkı (kâhin + kanıt + belgeler dışında BOŞ olmalı): [$(git diff $ESKI --name-only -- . ':!docs' ':!CLAUDE.md' ':!test/bagimsiz' | tr '\n' ' ')]"
TZ=UTC timeout 2400 node tools/kanit.mjs kos c5a1fy2-yeni-UTC -- node --disable-warning=ExperimentalWarning --test --test-concurrency=1 $YENI > $K/yeni-UTC.out 2>&1; echo "yeni testler UTC çıkış $? $(say $K/yeni-UTC.out)"
TZ=UTC timeout 3000 node test/bagimsiz/cari/kos.mjs --yapim-asamasi-yaz > $K/liste-uret.out 2>&1; echo "liste üretimi çıkış $? · fark (BOŞ olmalı): [$(git status -s -- $L)] $(grep -E 'ESKİ LİSTEYLE' $K/liste-uret.out | cut -c1-160)"
git diff -- $L > $K/liste-fark.diff; git checkout -q -- $L
grep -oE "kanit\.mjs kos kahin-[A-Za-z0-9_-]+ .*$" .github/workflows/ci.yml | sed 's/^kanit\.mjs kos //' > $K/kahin-liste.txt
echo "ci.yml'deki kâhin adımı sayısı $(wc -l < $K/kahin-liste.txt)"
while read -r ad komut; do
  case "$ad" in kahin-cari|kahin-defter) tzs="UTC Europe/Istanbul";; *) tzs="UTC";; esac
  for tz in $tzs; do m=${tz//\//-}
    TZ=$tz timeout 3000 node tools/kanit.mjs kos c5a1fy2-$ad-$m -- $komut > "$K/$ad-$m.out" 2>&1 < /dev/null; c=$?
    echo "$c $ad $tz :: $(grep -E '^\[kanit\]' "$K/$ad-$m.out" | tail -1 | cut -c1-150) $(grep -E 'ÖZET' "$K/$ad-$m.out" | tail -1 | cut -c1-150)"
  done
done < $K/kahin-liste.txt
for y in .github/workflows/ci.yml .github/workflows/release.yml; do python3 -I -c "import yaml,sys; yaml.safe_load(open(sys.argv[1]))" $y && echo "$y ayrıştı"; done
timeout 900 node --test test/yapim-asamasi-kapisi.test.mjs test/platform-tasinabilirlik.test.mjs test/kanit-araci.test.mjs test/ci-kapi-listesi.test.mjs > $K/dort-test.out 2>&1; echo "dört kapı testi çıkış $? $(say $K/dort-test.out)"
echo "kâhin bölümü bitti $(date -u +%T) yük $(cut -d' ' -f1-3 /proc/loadavg) · temiz mi: $(git status -s | grep -v node_modules | wc -l)"
flock -n $S/tam-npm-test.lock timeout 5400 node tools/kanit.mjs kos c5a1fy2-tam -- npm test > $K/tam.out 2>&1; echo "TAM npm test çıkış $? $(grep -E '^\[kanit\]' $K/tam.out | tail -1 | cut -c1-170)"; say $K/tam.out; echo
grep -E "^not ok|✖" $K/tam.out | sort -u | head -12 | cut -c1-220
echo "bitti $(date -u +%T) yük $(cut -d' ' -f1-3 /proc/loadavg) · aday $(git rev-parse --short HEAD) · temiz mi: $(git status -s | grep -v node_modules | wc -l) · disk $(df -h / | awk 'NR==2{print $4}')"
