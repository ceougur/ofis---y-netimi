#!/bin/bash
# Ana oturum: senaryo-220 — gövdeye eklenen öğelerin ZAMAN ÇİZELGESİ (izleme ön yüklemesi + tanı kopyası); ANA DAL KODUNDA ve ADAYDA.
# Çağrı: flock -s -n olcum.lock timeout 1500 bash s220-izle.sh <etiket> [GECIKTIR ms]
S=/tmp/claude-0/-home-user/de5e1cbf-7e68-5683-b7bb-f27655eaed4f/scratchpad
E=${1:-dogal}; G=${2:-0}
K=$S/c5a1fy-kanit2/s220-izle
export HUKUK_LICENSE_URL=http://127.0.0.1:9 PLAYWRIGHT_BROWSERS_PATH=$S/pw-1194 GECIKTIR=$G
mkdir -p $K || exit 1
echo "başladı $(date -u +%T) yük $(cut -d' ' -f1-3 /proc/loadavg) · etiket $E · geciktir $G"
for ad in ana aday; do
  case $ad in aday) W=$S/d2t3-ya-ag;; ana) W=$S/kahin8y-ag;; esac
  cp $S/senaryo-220-tani.mjs $W/test/e2e/senaryo-220-tani.mjs || exit 1
  cp $S/govde-izle.mjs $W/test/e2e/govde-izle-tani.mjs || exit 1
  ( cd $W && timeout 600 node --disable-warning=ExperimentalWarning --import ./test/e2e/govde-izle-tani.mjs test/e2e/senaryo-220-tani.mjs > $K/$E-$ad.out 2>&1; echo "$ad ($(git rev-parse --short HEAD)) çıkış $? · yük $(cut -d' ' -f1 /proc/loadavg)" )
  grep -E "govde-izle" $K/$E-$ad.out | grep -vE "\+ (a|input|iframe)\b" | head -40 | cut -c1-330
  grep -E "beklenmeyen hata|geçti, .* kaldı|waiting for" $K/$E-$ad.out | head -4 | cut -c1-300
  grep -E "TANI" $K/$E-$ad.out | cut -c1-900
  rm $W/test/e2e/senaryo-220-tani.mjs $W/test/e2e/govde-izle-tani.mjs
done
echo "bitti $(date -u +%T) · aday ağacı temiz mi: $(git -C $S/d2t3-ya-ag status -s | grep -v node_modules | wc -l) · ana kod ağacı: $(git -C $S/kahin8y-ag status -s | grep -v node_modules | wc -l)"
