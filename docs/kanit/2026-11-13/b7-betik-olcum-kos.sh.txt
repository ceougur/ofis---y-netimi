#!/bin/bash
# Birleşik tur 7 HIZ KABUL: özel kilitle çağrılır (flock -x -n olcum.lock). Taban ↔ yeni, iki kip, her ölçüm taze kopyada; yük > 2 ise satır GEÇERSİZ.
S=/tmp/claude-0/-home-user/de5e1cbf-7e68-5683-b7bb-f27655eaed4f/scratchpad
W=/home/user/ofis---y-netimi/.claude/worktrees/agent-a6c349be9d3823fef
export HUKUK_LICENSE_URL=http://127.0.0.1:9
OUT=$S/b7-kanit/b7-hiz.txt
K=$S/b7-olcum-kopya
echo "HIZ KABUL birleşik tur 7 · nproc $(nproc) · başta $(cat /proc/loadavg) · $(date -u +%H:%M:%S)" > $OUT
for tur in 1 2; do
  for kip in acilissiz acilisli; do
    for kod in taban:$S/b7-taban yeni:$W; do
      rm -rf $K && cp -a $S/ana-kabul/temiz-100k $K || exit 1
      sync; l0=$(cut -d' ' -f1 /proc/loadavg)
      echo "=== tur $tur · $kip · ${kod%%:*} · başta $(cat /proc/loadavg) $(awk -v l=$l0 'BEGIN{print (l>2)?"· GEÇERSİZ (yük > 2)":""}')" >> $OUT
      timeout -k 5 1500 node --disable-warning=ExperimentalWarning $S/b7-is/betik/olcum-b7.mjs ${kod#*:} $K $kip 5 >> $OUT 2>&1; echo "çıkış $? · sonda $(cat /proc/loadavg)" >> $OUT
    done
  done
done
rm -rf $K
echo "bitti $(date -u +%H:%M:%S)" >> $OUT
