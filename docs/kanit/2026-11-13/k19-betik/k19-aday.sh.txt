#!/bin/bash
# Ana oturum: KÂHİN PAKETİ (cari kâhini bakımı 17 + 18 + 19 + mobil yazma kâhini bakımı 2 + 3) — ADAY (kâhin ağacımda, dal sinama-k18 = çalışma dalı ucu + bakım 19 dalı + iki eşleme sınama commit'i; ön kontrolün yeniden ürettiği liste commit'siz duruyor — burada commit edilir).
# Liste yeniden üretimi + AYRIŞTIRMA (paket etiketleri) + mobil yazma kâhininin kabul dosyası; commit; kâhin öz testleri iki TZ; ci.yml'deki BÜTÜN kâhin adımları; dört kapı testi.
# Hata durdurmaz (kurulum adımları dışında); her adımın çıkışı yazılır.
S=/tmp/claude-0/-home-user/de5e1cbf-7e68-5683-b7bb-f27655eaed4f/scratchpad
W=$S/kahin8y-ag; K=$S/k19-aday-kanit; L=test/bagimsiz/cari/YAPIM-ASAMASI.json; MK=test/bagimsiz/mobil-yazma
export KANIT_DIR=$K HUKUK_LICENSE_URL=http://127.0.0.1:9 PLAYWRIGHT_BROWSERS_PATH=$S/pw-1194
G="git -c user.name=ana -c user.email=ana@yerel"
mkdir -p $K || exit 1
cd $W || exit 1
[ "$(git rev-parse --abbrev-ref HEAD)" = "sinama-k18" ] || { echo "kâhin ağacı sinama-k18 dalında değil"; exit 1; }
[ -e node_modules ] || ln -s /home/user/ofis---y-netimi/node_modules node_modules
echo "başladı $(date -u +%T) yük $(cut -d' ' -f1-3 /proc/loadavg) · ağaç $(git rev-parse --short HEAD) · ana dalla ürün farkı (kâhin + kanıt + iki eşleme dışında BOŞ olmalı): [$(git diff claude/epic-thompson-w0lc2l --name-only -- . ':!docs' ':!test/bagimsiz' | tr '\n' ' ')]"
echo "ön kontrolün bıraktığı liste: $(python3 -I -c "import json,sys; d=json.load(open(sys.argv[1],encoding='utf-8')); print(len(d['kayitlar']), 'kayıt')" $L)"
cp $MK/kabul.json $K/kabul-oncesi.json
timeout 1800 python3 -I -B $MK/senaryolar.py > $K/kabul-uret.out 2>&1; echo "mobil kabul üretimi çıkış $? :: $(tail -1 $K/kabul-uret.out | cut -c1-120)"
python3 -I -B docs/kanit/2026-11-13/kahinmobil2/arac/kabul-fark.py $K/kabul-oncesi.json $MK/kabul.json > $K/kabul-fark.out 2>&1; echo "kabul farkı çıkış $? :: $(tail -4 $K/kabul-fark.out | cut -c1-200 | tr '\n' '|')"
echo "mobil kabul dosyası değişti mi (BOŞ olmalı): [$(git status -s -- $MK/kabul.json)]"; git checkout -q -- $MK/kabul.json; git add $L && $G commit -q --no-verify -m "SINAMA: liste yeniden üretildi (gönderilmez)"; echo "aday commit $(git rev-parse --short HEAD) · temiz mi: $(git status -s | grep -v node_modules | wc -l)"
# liste yeniden üretilince etiketler korunuyor mu + kapı testleri
TZ=UTC timeout 3000 node test/bagimsiz/cari/kos.mjs --yapim-asamasi-yaz > $K/liste-uret2.out 2>&1; echo "liste 2. üretim çıkış $? · fark (BOŞ olmalı): [$(git status -s -- $L)] $(grep -E 'ESKİ LİSTEYLE' $K/liste-uret2.out | cut -c1-160)"
git checkout -q -- $L
for tz in UTC Europe/Istanbul; do m=${tz//\//-}
  TZ=$tz timeout 1800 node tools/kanit.mjs kos k19-cari-oz-$m -- python3 -I -B test/bagimsiz/cari/oz-test.py > $K/cari-oz-$m.out 2>&1; echo "cari öz test $tz çıkış $? :: $(grep -E '^\[kanit\]' $K/cari-oz-$m.out | tail -1 | cut -c1-130)"
  TZ=$tz timeout 1800 node tools/kanit.mjs kos k19-mobil-oz-$m -- python3 -I -B $MK/kahin_test.py > $K/mobil-oz-$m.out 2>&1; echo "mobil öz test $tz çıkış $? :: $(grep -E '^\[kanit\]' $K/mobil-oz-$m.out | tail -1 | cut -c1-130)"
done
grep -oE "kanit\.mjs kos kahin-[A-Za-z0-9_-]+ .*$" .github/workflows/ci.yml | sed 's/^kanit\.mjs kos //' > $K/kahin-liste.txt
echo "ci.yml'deki kâhin adımı sayısı $(wc -l < $K/kahin-liste.txt)"
while read -r ad komut; do
  case "$ad" in kahin-cari|kahin-defter) tzs="UTC Europe/Istanbul";; *) tzs="UTC";; esac
  for tz in $tzs; do m=${tz//\//-}
    TZ=$tz timeout 3000 node tools/kanit.mjs kos k19-$ad-$m -- $komut > "$K/$ad-$m.out" 2>&1 < /dev/null; c=$?
    echo "$c $ad $tz :: $(grep -E '^\[kanit\]' "$K/$ad-$m.out" | tail -1 | cut -c1-140) $(grep -E 'ÖZET' "$K/$ad-$m.out" | tail -1 | cut -c1-170)"
  done
done < $K/kahin-liste.txt
timeout 900 node --test test/yapim-asamasi-kapisi.test.mjs test/platform-tasinabilirlik.test.mjs test/kanit-araci.test.mjs test/ci-kapi-listesi.test.mjs > $K/dort-test.out 2>&1; echo "dört kapı testi çıkış $? $(grep -E '^# (tests|pass|fail)' $K/dort-test.out | tr '\n' ' ')"
echo "bitti $(date -u +%T) yük $(cut -d' ' -f1-3 /proc/loadavg) · aday $(git rev-parse --short HEAD) · temiz mi: $(git status -s | grep -v node_modules | wc -l)"
