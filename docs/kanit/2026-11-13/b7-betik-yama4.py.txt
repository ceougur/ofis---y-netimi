import sys
root = sys.argv[1]
def patch(rel, pairs):
    p = root + '/' + rel
    t = open(p, encoding='utf-8').read()
    for old, new in pairs:
        assert t.count(old) == 1, (rel, old[:90], t.count(old))
        t = t.replace(old, new)
    open(p, 'w', encoding='utf-8').write(t)
