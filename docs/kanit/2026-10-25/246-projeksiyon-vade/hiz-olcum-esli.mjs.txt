// 2.4.6 madde 11 eşli ölçüm: A (v2.4.5 gerçek kodu) ve B (bu dal) AYNI süreçte, ayrı veri kopyalarında açık; her turda iki istek dönüşümlü sırayla (ABAB / BABA),
// eş farkı (B − A) kaydedilir — makine yükünün kayması iki değişkene aynı biner. Kontrol: Tahsilat Takvimi (dues; iki sürümde aynı kod → Δ ≈ 0 beklenir).
//   REPO=<dal> node olcum-esli.mjs <kök> [tur=40]
import { cpSync, readFileSync, rmSync, writeFileSync } from "node:fs";
import path from "node:path";
const [rootArg, toursArg = "40"] = process.argv.slice(2);
const root = path.resolve(rootArg);
const REPO = process.env.REPO;
const { createClock } = await import(`${REPO}/server/lib/clock.mjs`);
const { bootVersion, CURRENT } = await import(`${REPO}/test/guvenilirlik/surumler.mjs`);
const manifest = JSON.parse(readFileSync(path.join(root, "manifest.json"), "utf8"));
const day = manifest.endDay;
const tours = Number(toursArg);
const median = list => { const s = [...list].sort((a, b) => a - b); return s.length % 2 ? s[(s.length - 1) / 2] : (s[s.length / 2 - 1] + s[s.length / 2]) / 2; };
const r1 = v => Math.round(v * 10) / 10;
const open = async (version, id) => {
  const dir = path.join(root, `esli-${id}`);
  rmSync(dir, { recursive: true, force: true });
  cpSync(path.join(root, "vadeli"), dir, { recursive: true });
  const clock = createClock({ time: new Date(Number(day.slice(0, 4)), Number(day.slice(5, 7)) - 1, Number(day.slice(8, 10)), 12, 0, 0), fixed: true });
  const extra = version === CURRENT ? { legacyCasePayments: false, legacyTableTransfer: false, legacyTableDues: false, moneyStrict: false, gateVerify: false } : {};
  const server = await bootVersion(version, { dataDir: path.join(dir, "data"), backupDir: path.join(dir, "backups"), env: { HUKUK_LICENSE_URL: "http://127.0.0.1:9" }, now: clock, edocEnabled: true, edocLive: true, esmmEnabled: true, ...extra });
  const api = await server.login();
  const timed = async url => { const t = performance.now(); const res = await api.client.raw("GET", url); if (res.status !== 200) throw new Error(`${id} ${url} ${res.status}`); return performance.now() - t; };
  return { id, timed, close: async () => { await server.close(); rmSync(dir, { recursive: true, force: true }); } };
};
const A = await open("v2.4.5", "A");
const B = await open(CURRENT, "B");
const out = {};
try {
  for (const [kind, url] of [["flow", "/api/workspace/overview/nakit-akisi?preset=next30"], ["dues", "/api/workspace/dues"]]) {
    await A.timed(url); await B.timed(url); // ısınma
    const a = []; const b = []; const d = [];
    for (let i = 0; i < tours; i += 1) {
      const order = i % 2 ? [B, A] : [A, B];
      const times = {};
      for (const side of order) times[side.id] = await side.timed(url);
      a.push(times.A); b.push(times.B); d.push(times.B - times.A);
    }
    const sorted = [...d].sort((x, y) => x - y);
    out[kind] = { tours, medianA: r1(median(a)), medianB: r1(median(b)), medianDelta: r1(median(d)), p25Delta: r1(sorted[Math.floor(tours * 0.25)]), p75Delta: r1(sorted[Math.floor(tours * 0.75)]), relative: `${r1((median(d) / median(a)) * 100)}%` };
    console.log(`${kind}: A ${out[kind].medianA} ms · B ${out[kind].medianB} ms · eş farkı (B − A) ortanca ${out[kind].medianDelta} ms [çeyrekler ${out[kind].p25Delta} … ${out[kind].p75Delta}] · ${out[kind].relative}`);
  }
} finally {
  await A.close(); await B.close();
}
writeFileSync(path.join(root, "olcum-esli-246.json"), `${JSON.stringify({ day, manifest: manifest.counts, vade: manifest.vade, out }, null, 1)}\n`);
console.log(`# tests 2\n# pass 2\n# fail 0\n# cancelled 0\n# skipped 0\n# todo 0`);
