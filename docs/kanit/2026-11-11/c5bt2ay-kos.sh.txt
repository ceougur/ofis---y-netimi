S=/tmp/claude-0/-home-user/de5e1cbf-7e68-5683-b7bb-f27655eaed4f/scratchpad

mkdir -p "$KANIT_DIR" "$E" "$M" || exit 1
export HUKUK_LICENSE_URL=http://127.0.0.1:9 PLAYWRIGHT_BROWSERS_PATH=$S/pw-1194
YENI="test/tyk-250.test.mjs test/tyk-onbellek-250.test.mjs test/tyk-tur2a-250.test.mjs test/tyk-api-250.test.mjs test/kapama-tek-gecis-245.test.mjs test/cheques-overview.test.mjs"

node tools/kanit.mjs kos c5bt2ay-ekran -- node --disable-warning=ExperimentalWarning test/e2e/senaryo-tyk-tur2a-250.mjs > "$KANIT_DIR/ekran.out" 2>&1; echo "ekran yeni çıkış $? $(tail -1 "$KANIT_DIR/ekran.out" | grep -oP '\d+/\d+ geçti, \d+ başarısız')"

node tools/kanit.mjs kos c5bt2ay-kahin-defter -- npm run test:kahin-defter > "$KANIT_DIR/kahin-defter.out" 2>&1; echo "kahin-defter çıkış $?"
