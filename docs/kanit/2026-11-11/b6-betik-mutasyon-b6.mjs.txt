const run = tests => {

  const out = `${r.stdout}\n${r.stderr}`;
  const count = key => Number((out.match(new RegExp(`^# ${key} (\\d+)$`, "m")) || [])[1] ?? NaN);
  return { status: r.status, out, pass: count("pass"), fail: count("fail"), skipped: count("skipped") };
};
const firstRed = out => {
  const name = (out.match(/^\s*not ok \d+ - (.*)$/m) || [])[1] || "";
  const diff = (out.match(/^\s+[+-] {2,}\S.*$/gm) || []).slice(0, 4).map(line => line.trim()).join(" | ");
  return `${name.slice(0, 110)}${diff ? ` :: ${diff.slice(0, 260)}` : ""}`;
};

    console.log(`taban ${key}: ${b.pass} geçti / ${b.fail} kaldı / ${b.skipped} atlandı (çıkış ${b.status})`);

  }
