const r2 = v => Math.round(v * 100) / 100;
const queue = acc => {
  if (!h.ctx.invoices.obligationsFor) return null;
  const res = h.ctx.invoices.obligationsFor(acc.id, h.ctx.accounts.detail(acc.id, AUD, { withRemaining: false }).ledger);
  return {
    kalemler: res.items.map(i => `${i.kind}|${i.side === "receivable" ? "+" : "-"}|${i.vade}|${i.amount}→${i.remaining}`),
    sigma: r2(res.items.reduce((s, i) => s + (i.side === "receivable" ? 1 : -1) * i.remaining, 0)),
    gecikmis: res.overdue(TODAY).amount,
    kapsama: [...(res.coverage || new Map()).entries()].map(([, v]) => `${v.rule}:tük${v.consumed}/kap${v.capacity}:avans${v.advance}:kırp${v.trimmed}`),
  };
};
const full = async (acc, opt = {}) => {
  const v = await views(h, acc, opt);
  const card = await h.card(acc);
  v.kalanNot = (card.entries || []).filter(e => e.remaining !== undefined).map(e => `${e.note || e.kind}:${e.amount}→${e.remaining}`);
  v.kuyruk = queue(acc);
  const flow = await h.api.get(`/api/workspace/overview/nakit-akisi?from=${day(-60)}&to=${day(60)}&overdue=1&table=0`);
  const rows = [...(flow.data?.rows || []), ...(flow.data?.overdue || [])].filter(r => r.party === acc.name || r.ref?.id === acc.id || r.accountId === acc.id);
  v.nakit = rows.map(r => `${r.source}:${r.label}:${r.direction}:${r.amount}@${r.date}`).sort();
  if (opt.plans) {
    v.kartAyrinti = [];
    for (const id of opt.plans) {
      const p = (await h.api.get(`/api/workspace/plans/${id}`)).data;
      v.kartAyrinti.push({ durum: p.state, taksitKalan: (p.items || []).map(i => i.remaining), taksitDurum: (p.items || []).map(i => i.state), kendiKalan: p.totals?.remaining, kuyrukKalan: p.totals?.queueRemaining, kendiOdenen: p.totals?.paid, cariPayi: p.totals?.accountPaid, geciken: p.totals?.overdue, gecikenAdet: p.totals?.overdueCount, gecikenRest: p.totals?.overdueRest, onerilen: p.ownNext?.remaining, kuyrukSiradaki: p.next?.remaining });
    }
  }
  return v;
};
const sc = async (name, fn) => {

    out[name] = { HATA: String(error?.message || error).slice(0, 500) };

{

  for (const [label, def] of Object.entries(PRE)) {
    pre[label] = { acc: await h.account(label, def.type || "customer"), inv: [] };
    def.invoices.forEach((row, index) => jobs.push({ label, index, row }));
  }
  jobs.sort((a, b) => a.row[0] - b.row[0]);
  for (const job of jobs) {
    const [d, amount, due, payment] = job.row;
    pre[job.label].inv[job.index] = await h.sale(pre[job.label].acc, day(d), amount, day(due), payment || null);
  }
}
