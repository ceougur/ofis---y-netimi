const snap = async (label, acc, planId, invoices = []) => {
  const v = await views(h, acc, { invoices, plans: [planId] });
  const card = await h.card(acc);
  const res = h.ctx.invoices.obligationsFor(acc.id, h.ctx.accounts.detail(acc.id, AUD, { withRemaining: false }).ledger);
  const sigma = r2(res.items.reduce((s, i) => s + (i.side === "receivable" ? 1 : -1) * i.remaining, 0));
  const kalan = (card.entries || []).filter(e => e.remaining !== undefined).map(e => `${e.note || e.label || e.kind}:${e.amount}→${e.remaining}`).join(", ");
  const k = v.kartlar[0];
  const out = { bakiye: v.bakiye, kartKendi: k.own, kartKuyruk: k.queue, taksit: k.items.join("/"), kartDurum: k.state, gecikmis: v.gecikmisKart, gecikmisListe: v.gecikmisListe, vadeGecikmis: v.vadeGecikmis, zil: v.zil.join(" ; "), vade: v.vade.join(" ; "), kalan, fatura: v.faturalar.map(f => `${f.state}:${f.open}`).join(", "), sigma };
  say(`${label}: bakiye ${out.bakiye} | kart kendi ${out.kartKendi} kuyruk ${out.kartKuyruk} [${out.taksit}] ${out.kartDurum} | gecikmiş ${out.gecikmis} = liste ${out.gecikmisListe} = Vade Takip ${out.vadeGecikmis} | zil {${out.zil}} | Kalan {${kalan}} | fatura {${out.fatura}} | Σ kalem ${sigma}`);
  return out;
};
