// Fikstürü verilen kodla açar, v249.json ile farkları döker (beklenen-farklar.json taslağı için). Kullanım: ROOT=<kod> node fark-dok.mjs <fikstür klasörü> <çıktı.json>
import { mkdirSync, mkdtempSync, readFileSync, writeFileSync, copyFileSync } from "node:fs";
import { tmpdir } from "node:os";
import path from "node:path";
import { pathToFileURL } from "node:url";
import { gunzipSync } from "node:zlib";

const ROOT = process.env.ROOT;
const [dir, outFile] = process.argv.slice(2);
const { startTestServer, loginAdmin } = await import(pathToFileURL(path.join(ROOT, "test/helpers.mjs")).href);
const { readFacts, diffFacts } = await import(pathToFileURL(path.join("/home/user/ofis---y-netimi/.claude/worktrees/agent-a1f2f9f23b4876417/test/tyk-fark-oku.mjs")).href);
const old = JSON.parse(readFileSync(path.join(dir, "v249.json"), "utf8"));
const root = mkdtempSync(path.join(tmpdir(), "fark-dok-"));
const dataDir = path.join(root, "data");
mkdirSync(dataDir, { recursive: true });
writeFileSync(path.join(dataDir, "destekofis.sqlite"), gunzipSync(readFileSync(path.join(dir, "destekofis.sqlite.gz"))));
copyFileSync(path.join(dir, "sirketler.json"), path.join(dataDir, "sirketler.json"));
const server = await startTestServer({ dataDir, backupDir: path.join(root, "backups"), now: { time: old.now, fixed: true }, env: { HUKUK_LICENSE_URL: "http://127.0.0.1:9" } });
const unwrap = r => (r.data && typeof r.data === "object" && "ok" in r.data ? (r.data.ok ? r.data.data : r.data) : r.data);
try {
  const client = await loginAdmin(server);
  const now = await readFacts(async url => { const r = await client.get(url); if (r.status !== 200) throw new Error(`${url} ${r.status}`); return unwrap(r); }, { today: old.today, prefix: "TYKF " });
  const diff = diffFacts(old, now);
  writeFileSync(outFile, JSON.stringify({ kod: ROOT, now, farklar: diff }, null, 1));
  for (const row of diff) console.log(`${row.cari} · ${row.alan}: ${row.eski} → ${row.yeni}`);
} finally {
  await server.close();
}
