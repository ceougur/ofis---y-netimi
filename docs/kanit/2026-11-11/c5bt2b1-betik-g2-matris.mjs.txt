// Tur 2b-1 G-2: kart türü (ayrı, Mevcut Borç, stoktan, faturadan, Excel) × taksit (yok / eksik / tam) × ödeme (yok / kart tahsilatı tam / kart seçilmeden cari tahsilatı tam)
// → kart durumu, Taksitler sekmeleri, taksit kalanları, Gecikmiş Borç. Üç kodda (v2.4.9, eski, yeni) aynı betik. Kullanım: ROOT=<kod> node g2-matris.mjs
import { boot, day, NOW, TODAY } from "./h.mjs";

const h = await boot({ now: { time: NOW, fixed: false } });
const out = { cases: {} };
const adv = () => h.server.clock.advance({ seconds: 5 });
try {
  const goods = await h.must("ürün", h.api.post("/api/workspace/stock", { name: "G2 Ürün", code: "URN-G2", unit: "Adet", unitPrice: "10", salePrice: "100" }));
  await h.must("alım", h.api.post(`/api/workspace/stock/${goods.id}/moves`, { kind: "in", qty: "1000", unitPrice: "10", pay: "none", date: day(-200) }));
  const planOf = async acc => (await h.card(acc)).plans?.[0]?.id || h.store.get("SELECT id FROM plans WHERE account_id = ? AND deleted_at IS NULL ORDER BY rowid DESC", acc.id)?.id;
  const trimTo = async (planId, keep) => {
    const p = (await h.api.get(`/api/workspace/plans/${planId}`)).data;
    const items = (p.items || []).slice(keep);
    const codes = [];
    for (const item of items) codes.push((await h.api.del(`/api/workspace/plans/${planId}/items/${item.id}`)).status);
    return codes.join(",");
  };
  const makers = {
    ayri: async (acc, mode) => (await h.must("kart", h.api.post("/api/workspace/plans", { name: acc.name, registeredOn: day(-30), total: "1000", accountId: acc.id, ...(mode === "tam" ? { mode: "auto", count: "2", firstDue: day(-10), everyMonths: "1" } : {}) }))).id,
    mevcut: async (acc, mode) => {
      await h.entry(acc, { kind: "debt", amount: "1000", date: day(-30) });
      adv();
      return (await h.must("kart", h.api.post("/api/workspace/plans", { name: acc.name, registeredOn: day(-30), total: "1000", accountId: acc.id, coversBalance: true, ...(mode === "tam" ? { mode: "auto", count: "2", firstDue: day(-10), everyMonths: "1" } : {}) }))).id;
    },
    stok: async acc => {
      await h.must("stok satış", h.api.post(`/api/workspace/stock/${goods.id}/moves`, { kind: "out", qty: "10", unitPrice: "100", pay: "account", method: "cash", accountId: acc.id, date: day(-30), installments: { count: 2, firstDue: day(-10) } }));
      return planOf(acc);
    },
    fatura: async acc => {
      await h.must("fatura", h.api.post("/api/workspace/invoices", { scenario: "service_sale", accountId: acc.id, issueDate: day(-30), lines: [{ itemId: h.item.id, qty: 1, unitPrice: "1000", vatRate: 0 }], payment: { rest: "installments", installments: { count: 2, firstDue: day(-10), everyMonths: 1 } }, cashForce: true }));
      return planOf(acc);
    },
    excel: async (acc, mode) => {
      const res = await h.api.post("/api/workspace/plans/import", { matrix: [["Ad Soyad", "Toplam", "Taksit", "İlk Vade", "Kayıt"], [acc.name, "1000", mode === "yok" ? "" : "2", day(-10).split("-").reverse().join("."), day(-30).split("-").reverse().join(".")]], headerAt: 0, roles: { 0: "name", 1: "total", 2: "count", 3: "firstDue", 4: "registered" }, fileName: `g2-${acc.name}.xlsx`, linkRecords: false });
      out.excelYukleme = out.excelYukleme || {};
      out.excelYukleme[acc.name] = res.status;
      const row = h.store.get("SELECT id, account_id AS acc FROM plans WHERE name = ? AND deleted_at IS NULL ORDER BY rowid DESC", acc.name);
      return row?.id ? { id: row.id, acc: row.acc } : null;
    },
  };
  const rows = [];
  for (const kind of Object.keys(makers)) {
    for (const mode of ["yok", "eksik", "tam"]) {
      for (const pay of ["yok", "kart", "cari"]) {
        const label = `G2 ${kind} ${mode} ${pay}`;
        let acc = kind === "excel" ? null : await h.account(label);
        let planId = null;
        let note = "";
        try {
          const made = await makers[kind](acc || { name: `DK ${label}` }, mode === "eksik" ? "tam" : mode);
          if (made && typeof made === "object") {
            planId = made.id;
            acc = h.store.get("SELECT id, name FROM accounts WHERE id = ?", made.acc);
          } else planId = made;
          adv();
          if (!planId) throw new Error("kart yok");
          if (mode === "eksik") note = `sil ${await trimTo(planId, 1)}`;
          if (mode === "yok" && ["stok", "fatura"].includes(kind)) note = `sil ${await trimTo(planId, 0)}`;
          adv();
          if (pay === "kart") note += ` kart ${(await h.api.post(`/api/workspace/plans/${planId}/entries`, { kind: "in", amount: "1000", date: day(-5), method: "cash", cashForce: true })).status}`;
          if (pay === "cari" && acc) note += ` cari ${(await h.api.post(`/api/workspace/accounts/${acc.id}/entries`, { kind: "in", amount: "1000", date: day(-5), method: "cash", cashForce: true })).status}`;
          adv();
        } catch (error) {
          note += ` HATA ${String(error.message).slice(0, 120)}`;
        }
        rows.push({ label, planId, acc, note });
      }
    }
  }
  const tabs = {};
  for (const status of ["active", "overdue", "done", "closed", "all"]) tabs[status] = new Set(((await h.api.get(`/api/workspace/plans?status=${status}&limit=5000`)).data?.plans || []).map(p => p.id));
  for (const row of rows) {
    if (!row.planId) {
      out.cases[row.label] = `kart yok · ${row.note}`;
      continue;
    }
    const p = (await h.api.get(`/api/workspace/plans/${row.planId}`)).data;
    const card = row.acc ? await h.card(row.acc) : { totals: {} };
    out.cases[row.label] = `${p.state} · sekme ${Object.entries(tabs).filter(([, s]) => s.has(row.planId)).map(([k]) => k).join("/")} · taksit [${(p.items || []).map(i => `${i.amount}:${i.remaining}:${i.state}`).join(" ")}] · bakiye ${card.totals.balance} · gecikmiş ${card.totals.overdueDebt ?? "-"}/${card.totals.overdue ?? "-"} ·${row.note}`;
  }
} finally {
  await h.server.close();
}
console.log(JSON.stringify(out, null, 1));
