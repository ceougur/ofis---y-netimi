import sys, pathlib
root = pathlib.Path(sys.argv[1]); name = sys.argv[2]
def edit(rel, old, new):
    p = root / rel
    s = p.read_text(encoding="utf-8")
    assert s.count(old) == 1, f"{name}: bulunamadı {old[:60]}"
    p.write_text(s.replace(old, new), encoding="utf-8")
