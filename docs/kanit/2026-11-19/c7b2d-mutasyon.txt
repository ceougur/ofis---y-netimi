with open(out, "w", encoding="utf-8") as log:
    for name, rel, old, new, test in [m for m in MUTANTS if not os.environ.get('YALNIZ') or m[0].startswith(os.environ['YALNIZ'])]:
        path = os.path.join(root, rel)
        src = open(path, encoding="utf-8").read()
        if src.count(old) != 1:
            log.write(f"{name}: UYGULANAMADI (desen {src.count(old)} kez)\n")
            continue
        open(path, "w", encoding="utf-8").write(src.replace(old, new))
        try:
            for tz in ("Europe/Istanbul", "UTC"):
                for run in (1, 2):
                    p = subprocess.run(["timeout", "-k", "5", "900", "node", "--test", test], cwd=root, env=dict(env, TZ=tz), capture_output=True, text=True)
                    text = p.stdout + p.stderr
                    counts = {k: next((l.split()[-1] for l in text.splitlines() if l.startswith(f"# {k} ")), "?") for k in ("tests", "pass", "fail", "cancelled")}
                    errors = [l.strip() for l in text.splitlines() if l.strip().startswith(("error:", "'", "+ ", "- ")) ][:3]
                    verdict = "YAKALANDI" if p.returncode != 0 and counts["fail"] not in ("0", "?") else "KAÇTI"
                    log.write(f"{name} | {tz} koşu {run}: çıkış {p.returncode} tests {counts['tests']} pass {counts['pass']} fail {counts['fail']} cancelled {counts['cancelled']} → {verdict} | {' / '.join(errors)[:300]}\n")
                    log.flush()
        finally:
            open(path, "w", encoding="utf-8").write(src)
    log.write("BİTTİ\n")
