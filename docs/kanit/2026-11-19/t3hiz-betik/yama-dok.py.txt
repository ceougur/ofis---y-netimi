    const D = await import(pathToFileURL(path.join(root, "server", "lib", "bank", "doubtful.mjs")).href);
    const st = ctx.store;
    const parties = st.all("SELECT DISTINCT account_id AS id FROM doubtful_receivables ORDER BY rowid").map(row => row.id);

      }
      lines.push(JSON.stringify({ sirket: company.id, cari: `[şüpheli ${pass}]`, tum: plain(D.doubtfulState(st)), tek: parties.map(id => plain(D.doubtfulState(st, { parties: [id] }))),
        acik: plain(D.doubtfulOpenByAccount(st)), acikTek: parties.map(id => plain(D.doubtfulOpenByAccount(st).byAccount.get(id) || null)),

    }
  }
