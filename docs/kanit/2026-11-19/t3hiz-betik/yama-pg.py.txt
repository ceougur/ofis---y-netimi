  const PERIOD_SUMS = `COALESCE(SUM(CASE WHEN :pfrom <> '' AND w.date < :pfrom THEN (CASE WHEN w.kind = 'in' THEN w.cents ELSE -w.cents END) ELSE 0 END), 0) AS opening,
        COALESCE(SUM(CASE WHEN (:pfrom = '' OR w.date >= :pfrom) AND w.kind = 'in' THEN w.cents ELSE 0 END), 0) AS debit,

  function periodGroups({ from = "", to = "" } = {}) {
