#!/bin/bash
# ANA OTURUM — TUR 3 + Ç7a + TUR 2f TEK ADAY HIZ KABUL (ders 15, 47, 52, 53 (a)). Taban = ana dal kodu (ayrık worktree), aday = sınama ağacı sinama-t3c7a.
# Her (betik, kip): taban → aday → taban → aday, her blok verinin TAZE kopyasında; ÖZEL ölçüm kilidi (tek seferlik bekler). Çağrı: timeout 7000 bash hiz.sh <kilitsiz|kilitli> [tekrar=5]
S=/tmp/claude-0/-home-user/de5e1cbf-7e68-5683-b7bb-f27655eaed4f/scratchpad; H=$S/ana-hiz; V=/dev/shm/ana-hiz
KIP=${1:-kilitsiz}; N=${2:-5}; YALNIZ=${3:-}; TAB=$H/taban; ADAY=$S/d2t3-ya-ag
export HUKUK_LICENSE_URL=http://127.0.0.1:9 TZ=Europe/Istanbul
exec 9>"$S/olcum.lock"; flock -x -w 3600 9 || { echo "KİLİT ALINAMADI — ÖLÇÜLEMEDİ"; exit 3; }
echo "kilit $(date -u +%H:%M:%S) yük $(cut -d' ' -f1-3 /proc/loadavg) · taban $(git -C $TAB rev-parse --short HEAD) · aday $(git -C $ADAY rev-parse --short HEAD) · $KIP"
for is in "t3b:$S/c5a3b-z4/betik/olc.mjs:$S/ana-kabul/temiz-100k" "t3c:$S/c5a3c-k7/olc-c5a3c.mjs:$V/hazir3c" "c7a:$S/c7a-t2b/olcum/olc-c7a3.mjs:$V/tohum7" "t2:$S/c5a2-olcum/olc-c5a2.mjs:$S/ana-kabul/temiz-100k"; do
  ad=${is%%:*}; r=${is#*:}; B=${r%%:*}; D=${r#*:}; OUT=$H/hiz-$ad-$KIP.txt
  [ -n "$YALNIZ" ] && [[ " $YALNIZ " != *" $ad "* ]] && continue
  echo "HIZ $ad · $KIP · tekrar $N · başta $(cat /proc/loadavg) · $(date -u +%H:%M:%S)" > $OUT
  for k in taban:$TAB aday:$ADAY taban:$TAB aday:$ADAY; do
    kn=${k%%:*}; R=${k#*:}; K=$V/kopya
    rm -rf $K && cp -a "$D" $K || exit 1
    [ "$KIP" = kilitli ] && { python3 -I $S/c5a2-olcum/kilit.py $K/data/destekofis.sqlite >> $OUT || exit 1; }
    sync; for w in $(seq 1 60); do awk '{exit !($1<1.5)}' /proc/loadavg && break; sleep 10; done  # ön koşul: hazırlığın (kopya + kilit) yükü sönsün; en çok 10 dk
    sync; l0=$(cut -d' ' -f1 /proc/loadavg)
    echo "=== $kn · başta $(cat /proc/loadavg) $(awk -v l=$l0 'BEGIN{print (l>2)?"· GEÇERSİZ (yük > 2)":""}')" >> $OUT
    if [ $ad = t3b ]; then
      timeout -k 5 2400 node --max-old-space-size=8192 --disable-warning=ExperimentalWarning $B "$R" $K $N $V/olc.json >> $OUT 2>&1; c=$?; python3 -I $H/json2satir.py $V/olc.json >> $OUT
    else
      timeout -k 5 2400 node --max-old-space-size=8192 --disable-warning=ExperimentalWarning $B "$R" $K $N >> $OUT 2>&1; c=$?
    fi
    echo "çıkış $c · sonda $(cat /proc/loadavg)" >> $OUT
  done
  rm -rf $V/kopya $V/olc.json
  echo "$ad bitti $(date -u +%H:%M:%S) · $(grep -c 'GEÇERSİZ' $OUT) geçersiz blok · çıkışlar: $(grep -o '^çıkış [0-9]*' $OUT | tr '\n' ' ')"
done
echo "bitti $(date -u +%H:%M:%S) yük $(cut -d' ' -f1-3 /proc/loadavg)"
