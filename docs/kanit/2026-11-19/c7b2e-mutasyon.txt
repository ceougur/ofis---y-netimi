def run(test, tz, tag):
    env = dict(os.environ, TZ=tz, HUKUK_LICENSE_URL="http://127.0.0.1:9")
    path = os.path.join(OUT, f"{tag}-{tz.replace('/', '-')}.out")
    with open(path, "w") as fh:
        p = subprocess.run(["timeout", "-k", "5", "900", "node", "--test", test], cwd=ROOT, env=env, stdout=fh, stderr=subprocess.STDOUT)
    text = open(path, encoding="utf-8", errors="replace").read()
    nums = {k: int(v) for k, v in re.findall(r"^# (tests|pass|fail|cancelled|skipped) (\d+)", text, re.M)}
    first = re.search(r"error: (?:\|-\n\s+)?'?(.{0,200})", text)
    return {"exit": p.returncode, **nums, "ilk": first.group(1).strip() if first else ""}

results = {}
for name, rel, old, new, tests in MUTANTS:
    file = os.path.join(ROOT, rel)
    src = open(file, encoding="utf-8").read()
    if src.count(old) != 1:
        results[name] = {"hata": f"desen {src.count(old)} kez"}
        continue
    open(file, "w", encoding="utf-8").write(src.replace(old, new))
    try:
        results[name] = {}
        for test in tests:
            for tz in ("Europe/Istanbul", "UTC"):
                tag = f"{name}-{os.path.basename(test).split('.')[0]}"
                results[name][f"{os.path.basename(test)} {tz}"] = run(test, tz, tag)
    finally:
        open(file, "w", encoding="utf-8").write(src)
    print(name, json.dumps(results[name], ensure_ascii=False), flush=True)
json.dump(results, open(os.path.join(OUT, "ozet.json"), "w"), ensure_ascii=False, indent=1)
