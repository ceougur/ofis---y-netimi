const only = process.argv[2] ? new RegExp(process.argv[2]) : null;
writeFileSync(OUT, `# mutasyon koşusu ${new Date().toISOString()} kopya ${MUT} (temiz: ${PRISTINE})\n`, { flag: "a" });
let caught = 0;
let survived = 0;
for (const [name, file, from, to, tests] of mutations) {
  if (only && !only.test(name)) continue;
  const target = path.join(MUT, file);
  const text = readFileSync(target, "utf8");
  const count = text.split(from).length - 1;
  if (count !== 1) {
    appendFileSync(OUT, `${name}: UYGULANAMADI (eşleşme ${count})\n`);
    console.log(`${name}: UYGULANAMADI (${count})`);
    continue;
  }
  writeFileSync(target, text.replace(from, to));
  const run = spawnSync(process.execPath, ["--test", "--test-concurrency=1", ...tests], { cwd: MUT, env: { ...process.env, HUKUK_LICENSE_URL: "http://127.0.0.1:9" }, encoding: "utf8", maxBuffer: 64 * 1024 * 1024, timeout: 900_000 });
  copyFileSync(path.join(PRISTINE, file), target);
  const out = `${run.stdout}\n${run.stderr}`;
  const num = key => Number((out.match(new RegExp(`^# ${key} (\\d+)`, "m")) || [])[1] ?? NaN);
  const failedNames = [...out.matchAll(/^ {4}not ok \d+ - (.{0,90})/gm)].map(m => m[1]);
  const verdict = run.status !== 0 && num("fail") > 0 ? "YAKALANDI" : "SAĞ KALDI";
  if (verdict === "YAKALANDI") caught += 1;
  else survived += 1;
  const line = `${name}: ${verdict} (çıkış ${run.status}, tests ${num("tests")}, fail ${num("fail")}, cancelled ${num("cancelled")}) ${failedNames.slice(0, 3).join(" | ")}`;
  appendFileSync(OUT, `${line}\n`);
  console.log(line);
}
const diff = spawnSync("diff", ["-rq", "--exclude=node_modules", "--exclude=data", path.join(PRISTINE, "server"), path.join(MUT, "server")], { encoding: "utf8" });
const tail = `# özet: yakalandı ${caught}, sağ kaldı ${survived}; kopya geri koyma denetimi: ${diff.status === 0 ? "AYNI" : `FARKLI\n${diff.stdout}`}`;
appendFileSync(OUT, `${tail}\n`);
console.log(tail);
process.exit(survived === 0 ? 0 : 1);
